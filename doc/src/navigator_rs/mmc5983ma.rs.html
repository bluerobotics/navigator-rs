12const DEFAULT_SPI_DEVICE: &str = "/dev/spidev1.1";
13const SPI_READ_FLAG: u8 = 0x80;
14const PRODUCT_ID: u8 = 0x30;
15
16/// Output of a null field in 16 bit mode.
17const ZERO_OFFSET: f32 = 32768.0;
18/// 4096 counts per gauss in 16 bit mode, one gauss being 100 µT.
19const COUNTS_TO_UT: f32 = 100.0 / 4096.0;
20/// How often the bridge offset is measured again. The offset drifts with
21/// temperature, and a SET/RESET pair costs two measurements.
22const DEGAUSS_INTERVAL: Duration = Duration::from_secs(10);
23
24/// Wrapper type for results
25pub type Result<T> = std::result::Result<T, Box<dyn Error>>;
26
27/// MMC5983MA Registers
28enum Register {
29    /// Magnetic field data registers
30    Xout0 = 0x00,
31    Status = 0x08,
32    Control0 = 0x09,
33    Control1 = 0x0A,
34    ProductId = 0x2F,
35}
36
37/// Bits of [`Register::Control0`]
38const CONTROL0_TAKE_MEASUREMENT: u8 = 0x01;
39const CONTROL0_SET: u8 = 0x08;
40const CONTROL0_RESET: u8 = 0x10;
41
42/// Maps the raw sensor axes into the board frame, x forward, y right, z down.
43///
44/// U15 sits rotated 90° on the bottom side of the board. This is the
45/// `ROTATION_YAW_180` that ArduPilot applies to this part on Navigator, and it
46/// goes with the sign that [`Mmc5983ma::degauss`] takes the difference in.
47fn to_board_frame(field: [f32; 3]) -> (f32, f32, f32) {
48    (-field[0], -field[1], field[2])
49}

113    pub fn build(self) -> Result<Mmc5983maDevice> {
114        let mut spi = Spidev::open(self.spi_device)?;
115        let options = SpidevOptions::new()
116            .bits_per_word(8)
117            .max_speed_hz(10_000_000)
118            .mode(SpiModeFlags::SPI_MODE_0)
119            .build();
120        spi.configure(&options)?;
121
122        let mut sensor = Mmc5983ma {
123            spi_device: spi,
124            offset: None,
125            last_degauss: Instant::now(),
126        };
127
128        sensor.begin()?;
129
130        Ok(Mmc5983maDevice {
131            mmc5983ma: sensor,
132            info: self.info,
133        })
134    }

143impl Mmc5983ma {
144    fn write_reg(&mut self, reg: Register, value: u8) -> Result<()> {
145        self.spi_device.write(&[reg as u8, value])?;
146        Ok(())
147    }
148
149    fn read_reg(&mut self, reg: Register) -> Result<u8> {
150        let mut frame = [reg as u8 | SPI_READ_FLAG, 0];
151        self.spi_device.transfer(&mut frame)?;
152        Ok(frame[1])
153    }
154
155    fn begin(&mut self) -> Result<()> {
156        // Reading the product ID over SPI fails on the first attempts, the
157        // part needs a few milliseconds before it answers
158        let mut product_id = 0;
159        for _ in 0..10 {
160            std::thread::sleep(Duration::from_millis(5));
161            product_id = self.read_reg(Register::ProductId)?;
162            if product_id == PRODUCT_ID {
163                break;
164            }
165        }
166        if product_id != PRODUCT_ID {
167            return Err(format!("MMC5983MA: Invalid product ID: 0x{:02X}", product_id).into());
168        }
169
170        self.write_reg(Register::Control1, 0x80)?;
171        std::thread::sleep(Duration::from_millis(15));
172
173        // Widest bandwidth, 100 Hz
174        self.write_reg(Register::Control1, 0x00)?;
175
176        self.degauss()?;
177
178        Ok(())
179    }
180
181    /// Takes a single measurement, in µT, without removing the bridge offset.
182    fn measure(&mut self) -> Result<[f32; 3]> {
183        self.write_reg(Register::Control0, CONTROL0_TAKE_MEASUREMENT)?;
184
185        let mut ready = false;
186        for _ in 0..20 {
187            std::thread::sleep(Duration::from_millis(1));
188            if self.read_reg(Register::Status)? & 0x01 != 0 {
189                ready = true;
190                break;
191            }
192        }
193        if !ready {
194            return Err("MMC5983MA: Timed out waiting for a measurement".into());
195        }
196
197        let mut frame = [0u8; 7];
198        frame[0] = Register::Xout0 as u8 | SPI_READ_FLAG;
199        self.spi_device.transfer(&mut frame)?;
200
201        Ok([
202            u16::from_be_bytes([frame[1], frame[2]]) as f32 - ZERO_OFFSET,
203            u16::from_be_bytes([frame[3], frame[4]]) as f32 - ZERO_OFFSET,
204            u16::from_be_bytes([frame[5], frame[6]]) as f32 - ZERO_OFFSET,
205        ]
206        .map(|counts| counts * COUNTS_TO_UT))
207    }
208
209    /// Drives the SET and RESET coils and measures on both, which flips the
210    /// field but not the bridge offset. Half the sum is then the offset and
211    /// half the difference the field, so this both measures and recalibrates.
212    ///
213    /// The difference is taken as reset minus set, the same way ArduPilot does
214    /// it, which inverts the field and is why [`to_board_frame`] undoes it.
215    fn degauss(&mut self) -> Result<[f32; 3]> {
216        let set = self.magnetize(CONTROL0_SET)?;
217        let reset = self.magnetize(CONTROL0_RESET)?;
218
219        let field = std::array::from_fn(|axis| (reset[axis] - set[axis]) * 0.5);
220        let offset: [f32; 3] = std::array::from_fn(|axis| (set[axis] + reset[axis]) * 0.5);
221
222        // Low pass the offset, a single pair is noisier than the field itself
223        self.offset = Some(match self.offset {
224            Some(previous) => std::array::from_fn(|axis| previous[axis] * 0.5 + offset[axis] * 0.5),
225            None => offset,
226        });
227
228        self.last_degauss = Instant::now();
229        Ok(field)
230    }
231
232    /// Drives one of the coils and measures once it has settled.
233    fn magnetize(&mut self, coil: u8) -> Result<[f32; 3]> {
234        self.write_reg(Register::Control0, coil)?;
235        // The datasheet asks for 1 ms between driving a coil and measuring
236        std::thread::sleep(Duration::from_millis(1));
237        self.measure()
238    }
239
240    fn read_field(&mut self) -> Result<[f32; 3]> {
241        if self.last_degauss.elapsed() >= DEGAUSS_INTERVAL {
242            return self.degauss();
243        }
244
245        let field = self.measure()?;
246        let offset = self.offset.unwrap_or_default();
247        Ok(std::array::from_fn(|axis| field[axis] - offset[axis]))
248    }
249}

251impl MagnetometerSensor for Mmc5983maDevice {
252    fn read_magnetic_field(&mut self) -> Result<(f32, f32, f32)> {
253        let field = self.mmc5983ma.read_field()?;
254        Ok(to_board_frame(field))
255    }
