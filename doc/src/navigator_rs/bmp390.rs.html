17/// BMP390 Registers
18enum Register {
19    ChipId = 0x00,
20    ErrReg = 0x02,
21    Event = 0x10,
22    /// Calibration data start
23    NvmParT1_0 = 0x31,
24    /// data registers
25    Data0 = 0x04, // pressure XLSB
26    Data3 = 0x07, // temperature XLSB
27
28    PwrCtrl = 0x1B,
29}

31/// Output from the BMP390 consists of ADC outputs.
32///
33/// These must be compensated using formulas from the datasheet to obtain the actual temperature and pressure values,
34/// using coefficients stored in non-volatile memory (NVM).
35///
36/// # Datasheet
37/// - Section 3.11 Output compensation.
38/// - Appendix A: Computation formulae reference implementation.
39#[derive(Debug, Default, Clone, Copy)]
40struct CalibrationCoefficients {
41    par_t1: f32,
42    par_t2: f32,
43    par_t3: f32,
44    par_p1: f32,
45    par_p2: f32,
46    par_p3: f32,
47    par_p4: f32,
48    par_p5: f32,
49    par_p6: f32,
50    par_p7: f32,
51    par_p8: f32,
52    par_p9: f32,
53    par_p10: f32,
54    par_p11: f32,
55}

95impl Bmp390Builder {
96    pub fn new() -> Self {
97        Bmp390Builder {
98            i2c_address: DEFAULT_I2C_ADDRESS,
99            i2c_path: PathBuf::from(DEFAULT_I2C_PATH),
100            ground_pressure: 0.,
101        }
102    }
103
104    pub fn address(&mut self, address: u16) -> &mut Self {
105        self.i2c_address = address as u8;
106        self
107    }
108
109    pub fn path(&mut self, path: impl Into<PathBuf>) -> &mut Self {
110        self.i2c_path = path.into();
111        self
112    }
113
114    pub fn ground_pressure(&mut self, pressure: f32) -> &mut Self {
115        self.ground_pressure = pressure;
116        self
117    }
118
119    pub fn build(&self) -> Result<Bmp390Device> {
120        let dev = I2cdev::new(&self.i2c_path)?;
121        let mut sensor = Bmp390 {
122            i2c_device: dev,
123            address: self.i2c_address,
124            calibration: CalibrationCoefficients::default(),
125            ground_pressure: self.ground_pressure,
126        };
127
128        sensor.begin()?;
129
130        if self.ground_pressure != 0. {
131            sensor.zero()?;
132        }
133
134        Ok(Bmp390Device {
135            bmp390: sensor,
136            info: PeripheralInfo {
137                peripheral: Peripherals::Bmp390,
138                class: vec![PeripheralClass::Pressure, PeripheralClass::Temperature],
139            },
140        })
141    }
142}

150impl Bmp390 {
151    fn write_reg(&mut self, reg: Register, value: u8) -> Result<()> {
152        self.i2c_device.write(self.address, &[reg as u8, value])?;
153        Ok(())
154    }
155
156    fn read_reg(&mut self, reg: Register) -> Result<u8> {
157        let mut buf = [0u8; 1];
158        self.i2c_device
159            .write_read(self.address, &[reg as u8], &mut buf)?;
160        Ok(buf[0])
161    }
162
163    fn read_burst(&mut self, start: Register, buf: &mut [u8]) -> Result<()> {
164        self.i2c_device
165            .write_read(self.address, &[start as u8], buf)?;
166        Ok(())
167    }
168
169    fn begin(&mut self) -> Result<()> {
170        let chip_id = self.read_reg(Register::ChipId)?;
171        if chip_id != 0x60 {
172            return Err(format!("BMP390: Invalid chip ID: 0x{:02X}", chip_id).into());
173        }
174
175        // reset device not strictly necessary here
176        // Configure sensor (similar to BMP390 defaults)
177        // Enable temperature and pressure, normal mode
178        // PwrCtrl: enable temp + pressure + normal mode: 0x33 (temp + press + normal)
179        self.write_reg(Register::PwrCtrl, 0x33)?;
180
181        // wait a bit for device to load NVM
182        std::thread::sleep(std::time::Duration::from_millis(2));
183
184        self.read_calibration()?;
185
186        Ok(())
187    }
188
189    pub fn zero(&mut self) -> Result<f32> {
190        self.ground_pressure = self.pressure_pa()?;
191        Ok(self.ground_pressure)
192    }
193
194    fn read_calibration(&mut self) -> Result<()> {
195        // Reading calibration coefficients from BMP390:
196        let mut data = [0u8; 21];
197        self.read_burst(Register::NvmParT1_0, &mut data)?;
198
199        // Parsing according to BMP390 datasheet (from earlier code):
200        let nvm_par_t1 = ((data[1] as u16) << 8) | data[0] as u16;
201        let nvm_par_t2 = ((data[3] as u16) << 8) | data[2] as u16;
202        let nvm_par_t3 = data[4] as i8;
203        let nvm_par_p1 = ((data[6] as i16) << 8) | data[5] as i16;
204        let nvm_par_p2 = ((data[8] as i16) << 8) | data[7] as i16;
205        let nvm_par_p3 = data[9] as i8;
206        let nvm_par_p4 = data[10] as i8;
207        let nvm_par_p5 = ((data[12] as u16) << 8) | data[11] as u16;
208        let nvm_par_p6 = ((data[14] as u16) << 8) | data[13] as u16;
209        let nvm_par_p7 = data[15] as i8;
210        let nvm_par_p8 = data[16] as i8;
211        let nvm_par_p9 = ((data[18] as i16) << 8) | data[17] as i16;
212        let nvm_par_p10 = data[19] as i8;
213        let nvm_par_p11 = data[20] as i8;
214
215        self.calibration = CalibrationCoefficients {
216            par_t1: (nvm_par_t1 as f32) / 0.00390625,
217            par_t2: (nvm_par_t2 as f32) / 1073741824.0,
218            par_t3: (nvm_par_t3 as f32) / 281474976710656.0,
219            par_p1: ((nvm_par_p1 as f32) - 16384.0) / 1048576.0,
220            par_p2: ((nvm_par_p2 as f32) - 16384.0) / 536870912.0,
221            par_p3: (nvm_par_p3 as f32) / 4294967296.0,
222            par_p4: (nvm_par_p4 as f32) / 137438953472.0,
223            par_p5: (nvm_par_p5 as f32) / 0.125,
224            par_p6: (nvm_par_p6 as f32) / 64.0,
225            par_p7: (nvm_par_p7 as f32) / 256.0,
226            par_p8: (nvm_par_p8 as f32) / 32768.0,
227            par_p9: (nvm_par_p9 as f32) / 281474976710656.0,
228            par_p10: (nvm_par_p10 as f32) / 281474976710656.0,
229            par_p11: (nvm_par_p11 as f32) / 36893488147419103232.0,
230        };
231
232        Ok(())
233    }
234
235    fn read_raw_measurements(&mut self) -> Result<(u32, u32)> {
236        // BMP390 data: pressure in DATA_0..2, temp in DATA_3..5
237        let mut buf = [0u8; 6];
238        self.read_burst(Register::Data0, &mut buf)?;
239
240        let raw_pressure = (buf[0] as u32) | ((buf[1] as u32) << 8) | ((buf[2] as u32) << 16);
241        let raw_temperature = (buf[3] as u32) | ((buf[4] as u32) << 8) | ((buf[5] as u32) << 16);
242
243        Ok((raw_temperature, raw_pressure))
244    }
245
246    fn temperature_celsius(&mut self) -> Result<f32> {
247        let (raw_temp, _) = self.read_raw_measurements()?;
248        let t_lin = self.compensate_temperature(raw_temp);
249        Ok(t_lin)
250    }
251
252    fn pressure_pa(&mut self) -> Result<f32> {
253        let (raw_temp, raw_press) = self.read_raw_measurements()?;
254
255        let t_lin = self.compensate_temperature(raw_temp);
256        let pressure = self.compensate_pressure(t_lin, raw_press);
257        Ok(pressure)
258    }
259
260    /// Compensate a temperature reading according to calibration coefficients.
261    ///
262    /// # Datasheet
263    /// Apendix A, Section 8.5
264    fn compensate_temperature(&self, raw_temp: u32) -> f32 {
265        let uncomp = raw_temp as f32;
266        let partial_data1 = uncomp - self.calibration.par_t1;
267        let partial_data2 = partial_data1 * self.calibration.par_t2;
268        partial_data2 + (partial_data1 * partial_data1) * self.calibration.par_t3
269    }
270
271    /// Compensate a pressure reading according to calibration coefficients.
272    ///
273    /// # Datasheet
274    /// Apendix A, Section 8.6
275    fn compensate_pressure(&self, temperature: f32, raw_press: u32) -> f32 {
276        let uncomp = raw_press as f32;
277
278        let partial_data1 = self.calibration.par_p6 * temperature;
279        let partial_data2 = self.calibration.par_p7 * temperature * temperature;
280        let partial_data3 = self.calibration.par_p8 * temperature * temperature * temperature;
281        let partial_out1 = self.calibration.par_p5 + partial_data1 + partial_data2 + partial_data3;
282
283        let partial_data1 = self.calibration.par_p2 * temperature;
284        let partial_data2 = self.calibration.par_p3 * temperature * temperature;
285        let partial_data3 = self.calibration.par_p4 * temperature * temperature * temperature;
286        let partial_out2 =
287            uncomp * (self.calibration.par_p1 + partial_data1 + partial_data2 + partial_data3);
288
289        let partial_data1 = uncomp * uncomp;
290        let partial_data2 = self.calibration.par_p9 + self.calibration.par_p10 * temperature;
291        let partial_data3 = partial_data1 * partial_data2;
292        let partial_data4 = partial_data3 + uncomp * uncomp * uncomp * self.calibration.par_p11;
293
294        partial_out1 + partial_out2 + partial_data4
295    }
296}
