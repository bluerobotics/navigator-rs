63macro_rules! match_channel {
64    ($value:expr, $($v:expr, $C:ident),*) => {
65        match $value {
66            $(
67                $v => Ok(Channel::$C),
68            )*
69            _ => Err(()),
70        }
71    };

179/// I2C device address
180#[derive(Debug, Copy, Clone, PartialEq)]
181pub struct Address(pub(crate) u8);
182
183/// Default device address
184impl Default for Address {
185    fn default() -> Self {
186        Address(DEVICE_BASE_ADDRESS)
187    }
188}
189
190/// Support custom (integer) addresses
191impl From<u8> for Address {
192    fn from(a: u8) -> Self {
193        Address(a)
194    }
195}
196
197/// Compute device address from address bits
198impl From<(bool, bool, bool, bool, bool, bool)> for Address {
199    fn from(a: (bool, bool, bool, bool, bool, bool)) -> Self {
200        Address(
201            DEVICE_BASE_ADDRESS
202                | ((a.0 as u8) << 5)
203                | ((a.1 as u8) << 4)
204                | ((a.2 as u8) << 3)
205                | ((a.3 as u8) << 2)
206                | ((a.4 as u8) << 1)
207                | a.5 as u8,
208        )
209    }
210}
