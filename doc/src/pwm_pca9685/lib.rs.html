1//! This is a platform agnostic Rust driver for the PCA9685 PWM/Servo/LED
2//! controller, based on the [`embedded-hal`] traits.
3//!
4//! [`embedded-hal`]: https://github.com/rust-embedded/embedded-hal
5//!
6//! This driver allows you to:
7//! - Enable/disable the device. See: [`enable()`](Pca9685::enable).
8//! - Set the _on_ and _off_ counter for a channel or all of them. See: [`set_channel_on()`](Pca9685::set_channel_on).
9//! - Set the _on_ and _off_ counters for a channel or all of them at once. See: [`set_channel_on_off()`](Pca9685::set_channel_on_off).
10//! - Set a channel to be always on or off. See: [`set_channel_full_on()`](Pca9685::set_channel_full_on).
11//! - Set the _on_ and _off_ counters for each channel at once. See: [`set_all_on_off()`](Pca9685::set_all_on_off).
12//! - Set the prescale value. See: [`set_prescale()`](Pca9685::set_prescale).
13//! - Select the output logic state direct or inverted. See: [`set_output_logic_state()`](Pca9685::set_output_logic_state).
14//! - Set when the outputs change. See: [`set_output_change_behavior()`](Pca9685::set_output_change_behavior).
15//! - Set the output driver configuration. See: [`set_output_driver()`](Pca9685::set_output_driver).
16//! - Set the output value when outputs are disabled. See: [`set_disabled_output_value()`](Pca9685::set_disabled_output_value)
17//! - Select the EXTCLK pin as clock source. See: [`use_external_clock()`](Pca9685::use_external_clock).
18//! - Enable/disable a programmable address. See: [`enable_programmable_address()`](Pca9685::enable_programmable_address).
19//! - Set a programmable address. See: [`set_programmable_address()`](Pca9685::set_programmable_address).
20//! - Change the address used by the driver. See: [`set_address()`](Pca9685::set_address).
21//! - Restart keeping the PWM register contents. See: [`enable_restart_and_disable()`](Pca9685::enable_restart_and_disable).
22//!
23//! [Introductory blog post](https://blog.eldruin.com/pca9685-pwm-led-servo-controller-driver-in-rust/)
24//!
25//! ## The device
26//!
27//! This device is an I2C-bus controlled 16-channel, 12-bit PWM controller.
28//! Its outputs can be used to control servo motors or LEDs, for example.
29//!
30//! Each channel output has its own 12-bit resolution (4096 steps) fixed
31//! frequency individual PWM controller that operates at a programmable
32//! frequency from a typical of 24 Hz to 1526 Hz with a duty cycle that is
33//! adjustable from 0% to 100%.
34//! All outputs are set to the same PWM frequency.
35//!
36//! Each channel output can be off or on (no PWM control), or set at its
37//! individual PWM controller value. The output driver is programmed to be
38//! either open-drain with a 25 mA current sink capability at 5 V or totem pole
39//! with a 25 mA sink, 10 mA source capability at 5 V. The PCA9685 operates
40//! with a supply voltage range of 2.3 V to 5.5 V and the inputs and outputs
41//! are 5.5 V tolerant. LEDs can be directly connected to the outputs (up to
42//! 25 mA, 5.5 V) or controlled with external drivers and a minimum amount of
43//! discrete components for larger current, higher voltage LEDs, etc.
44//! It is optimized to be used as an LED controller for Red/Green/Blue/Amber
45//! (RGBA) color backlighting applications.
46//!
47//! Datasheet: [PCA9685](https://www.nxp.com/docs/en/data-sheet/PCA9685.pdf)
48//!
49//! ## Usage examples (see also examples folder)
50//!
51//! To use this driver, import this crate and an `embedded_hal` implementation,
52//! then instantiate the appropriate device.
53//!
54//! Please find additional examples in this repository: [driver-examples]
55//!
56//! [driver-examples]: https://github.com/eldruin/driver-examples
57//!
58//! ### Create a driver instance
59//!
60//! ```no_run
61//! use linux_embedded_hal::I2cdev;
62//! use pwm_pca9685::{Address, Pca9685};
63//!
64//! let dev = I2cdev::new("/dev/i2c-1").unwrap();
65//! let address = Address::default();
66//! let pwm = Pca9685::new(dev, address).unwrap();
67//! // do something...
68//!
69//! // get the I2C device back
70//! let dev = pwm.destroy();
71//! ```
72//!
73//! ### Create a driver instance for the PCA9685 with an alternative address
74//!
75//! ```no_run
76//! use linux_embedded_hal::I2cdev;
77//! use pwm_pca9685::{Address, Pca9685};
78//!
79//! let dev = I2cdev::new("/dev/i2c-1").unwrap();
80//! let (a5, a4, a3, a2, a1, a0) = (false, true, false, true, true, false);
81//! let address = (a5, a4, a3, a2, a1, a0);
82//! let pwm = Pca9685::new(dev, address).unwrap();
83//! ```
84//!
85//! ### Set the PWM frequency and channel duty cycles
86//!
87//! - Set a PWM frequency of 60 Hz (corresponds to a value of 100 for the
88//!   prescale).
89//! - Set a duty cycle of 50% for channel 0.
90//! - Set a duty cycle of 75% for channel 1 delayed 814 µs with respect
91//!   to channel 0.
92//!
93//! ```no_run
94//! use linux_embedded_hal::I2cdev;
95//! use pwm_pca9685::{Address, Channel, Pca9685};
96//!
97//! let dev = I2cdev::new("/dev/i2c-1").unwrap();
98//! let address = Address::default();
99//! let mut pwm = Pca9685::new(dev, address).unwrap();
100//! pwm.set_prescale(100).unwrap();
101//! pwm.enable().unwrap();
102//!
103//! // Turn on channel 0 at 0 and off at 2047, which is 50% in the range `[0..4095]`.
104//! pwm.set_channel_on_off(Channel::C0, 0, 2047).unwrap();
105//!
106//! // Turn on channel 1 at 200, then off at 3271. These values comes from:
107//! // 0.000814 (seconds) * 60 (Hz) * 4096 (resolution) = 200
108//! // 4096 * 0.75 + 200 = 3272
109//! pwm.set_channel_on_off(Channel::C1, 200, 3272).unwrap();
110//! ```
111//!
112//! ### Set the PWM frequency and channel duty cycles separately
113//!
114//! - Set a PWM frequency of 60 Hz (corresponds to a value of 100 for the
115//!   prescale).
116//! - Set a duty cycle of 50% for channel 0.
117//! - Set a duty cycle of 75% for channel 1 delayed 814 µs with respect
118//!   to channel 0.
119//!
120//! ```no_run
121//! use linux_embedded_hal::I2cdev;
122//! use pwm_pca9685::{Address, Channel, Pca9685};
123//!
124//! let dev = I2cdev::new("/dev/i2c-1").unwrap();
125//! let address = Address::default();
126//! let mut pwm = Pca9685::new(dev, address).unwrap();
127//! pwm.set_prescale(100).unwrap();
128//! pwm.enable().unwrap();
129//!
130//! // Turn on channel 0 at 0
131//! pwm.set_channel_on(Channel::C0, 0).unwrap();
132//!
133//! // Turn off channel 0 at 2047, which is 50% in the range `[0..4095]`.
134//! pwm.set_channel_off(Channel::C0, 2047).unwrap();
135//!
136//! // Turn on channel 1 at 200. This value comes from:
137//! // 0.000814 (seconds) * 60 (Hz) * 4096 (resolution) = 200
138//! pwm.set_channel_on(Channel::C1, 200).unwrap();
139//!
140//! // Turn off channel 1 at 3271, which is 75% in the range `[0..4095]`
141//! // plus 200 which is when the channel turns on.
142//! pwm.set_channel_off(Channel::C1, 3271).unwrap();
143//! ```
144//!
145//! ### Set a channel completely on and off (beware of precedences).
146//!
147//! ```no_run
148//! use linux_embedded_hal::I2cdev;
149//! use pwm_pca9685::{Address, Channel, Pca9685};
150//!
151//! let dev = I2cdev::new("/dev/i2c-1").unwrap();
152//! let address = Address::default();
153//! let mut pwm = Pca9685::new(dev, address).unwrap();
154//! pwm.enable().unwrap();
155//!
156//! // Turn channel 0 full on at 1024
157//! pwm.set_channel_full_on(Channel::C0, 1024).unwrap();
158//!
159//! // Turn channel 0 full off (full off takes precedence over on settings)
160//! pwm.set_channel_full_off(Channel::C0).unwrap();
161//!
162//! // Return channel 0 to full on by deactivating full off.
163//! // The value is ignored because full on takes precedence
164//! // over off settings except full off.
165//! let value_ignored_for_now = 2048;
166//! pwm.set_channel_off(Channel::C0, value_ignored_for_now).unwrap();
167//!
168//! // Deactivate full on and set a duty cycle of 50% for channel 0.
169//! // (on from 0 to 2047, then off)
170//! pwm.set_channel_on(Channel::C0, 0).unwrap();
171//! ```
172//!
173//! ### Set a 50% duty cycle for all channels at once
174//!
175//! ```no_run
176//! use linux_embedded_hal::I2cdev;
177//! use pwm_pca9685::{ Channel, Pca9685, Address };
178//!
179//! let dev = I2cdev::new("/dev/i2c-1").unwrap();
180//! let address = Address::default();
181//! let mut pwm = Pca9685::new(dev, address).unwrap();
182//! pwm.enable().unwrap();
183//!
184//! let mut on = [0; 16];
185//! let mut off = [2047; 16];
186//! pwm.set_all_on_off(&on, &off);
187//! ```
188//!
189//! ### Use a programmable address
190//!
191//! Several additional addresses can be programmed for the device (they are
192//! volatile, though).
193//! Once set it is necessary to enable them so that the device responds to
194//! them. Then it is possible to change the address that the driver uses
195//! to communicate with the device.
196//!
197//! ```no_run
198//! use linux_embedded_hal::I2cdev;
199//! use pwm_pca9685::{Channel, Pca9685, Address, ProgrammableAddress};
200//!
201//! let dev = I2cdev::new("/dev/i2c-1").unwrap();
202//! let hardware_address = Address::default();
203//! let mut pwm = Pca9685::new(dev, hardware_address).unwrap();
204//!
205//! let subaddr1 = 0x71;
206//! pwm.set_programmable_address(ProgrammableAddress::Subaddress1, subaddr1).unwrap();
207//! pwm.enable_programmable_address(ProgrammableAddress::Subaddress1).unwrap();
208//!
209//! // Now communicate using the new address:
210//! pwm.set_address(subaddr1).unwrap();
211//! pwm.enable().unwrap();
212//! pwm.set_channel_on_off(Channel::C0, 0, 2047).unwrap();
213//!
214//! // The device will also respond to the hardware address:
215//! pwm.set_address(hardware_address).unwrap();
216//! pwm.set_channel_on_off(Channel::C0, 2047, 4095).unwrap();
217//!
218//! // when done you can also disable responding to the additional address:
219//! pwm.disable_programmable_address(ProgrammableAddress::Subaddress1).unwrap();
220//! ```
221//!
222//! ### Put the device to sleep then restart previously active PWM channels
223//!
224//! ```no_run
225//! use linux_embedded_hal::{Delay, I2cdev};
226//! use pwm_pca9685::{Address, Channel, Pca9685};
227//!
228//! let dev = I2cdev::new("/dev/i2c-1").unwrap();
229//! let mut pwm = Pca9685::new(dev, Address::default()).unwrap();
230//! pwm.enable().unwrap();
231//!
232//! pwm.set_channel_on_off(Channel::C0, 0, 2047).unwrap();
233//! // Prepare for restart and put the device to sleep
234//! pwm.enable_restart_and_disable().unwrap();
235//! // ...
236//! // re-enable device and reactivate channel 0
237//! let mut delay = Delay{};
238//! pwm.restart(&mut delay).unwrap();
239//! ```
