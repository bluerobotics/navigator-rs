42    /// Put the controller to sleep while keeping the PWM register
43    /// contents in preparation for a future restart.
44    pub fn enable_restart_and_disable(&mut self) -> Result<(), Error<E>> {
45        let config = self.config.with_high(BitFlagMode1::Sleep);
46        self.write_mode1(config.with_high(BitFlagMode1::Restart))?;
47        // Do not store restart bit high as writing this bit high again
48        // would internally clear it to 0. Writing 0 has no effect.
49        self.config = config;
50        Ok(())
51    }

53    /// Re-enable the controller after a sleep with restart enabled so that
54    /// previously active PWM channels are restarted.
55    ///
56    /// This includes a delay of 500us in order for the oscillator to stabilize.
57    /// If you cannot afford a 500us delay you can use `restart_nonblocking()`.
58    pub fn restart(&mut self, delay: &mut impl DelayUs<u16>) -> Result<(), Error<E>> {
59        let mode1 = self.read_register(Register::MODE1)?;
60        if (mode1 & BitFlagMode1::Restart as u8) != 0 {
61            self.enable()?;
62            delay.delay_us(500_u16);
63            let previous = self.config;
64            let config = previous.with_high(BitFlagMode1::Restart);
65            self.write_mode1(config)?;
66            self.config = previous;
67        }
68        Ok(())
69    }

71    /// Re-enable the controller after a sleep with restart enabled so that
72    /// previously active PWM channels are restarted (non-blocking version).
73    ///
74    /// This is a nonblocking version where you are responsible for waiting at
75    /// least 500us after the receiving the first `WouldBlock` error before
76    /// calling again to continue.
77    pub fn restart_nonblocking(&mut self) -> nb::Result<(), Error<E>> {
78        let mode1 = self
79            .read_register(Register::MODE1)
80            .map_err(nb::Error::Other)?;
81        let restart_high = (mode1 & BitFlagMode1::Restart as u8) != 0;
82        let sleep_high = (mode1 & BitFlagMode1::Sleep as u8) != 0;
83        if restart_high {
84            if sleep_high {
85                self.enable().map_err(nb::Error::Other)?;
86                return Err(nb::Error::WouldBlock);
87            } else {
88                let previous = self.config;
89                let config = previous.with_high(BitFlagMode1::Restart);
90                self.write_mode1(config).map_err(nb::Error::Other)?;
91                self.config = previous;
92            }
93        }
94        Ok(())
95    }

97    /// Set one of the programmable addresses.
98    ///
99    /// Initially these are not enabled. Once you set this, you can call
100    /// `enable_programmable_address()` and then use `set_address()` to configure
101    /// the driver to use the new address.
102    pub fn set_programmable_address<A: Into<Address>>(
103        &mut self,
104        address_type: ProgrammableAddress,
105        address: A,
106    ) -> Result<(), Error<E>> {
107        let a = address.into();
108
109        Self::check_address(a.0)?;
110        let reg = match address_type {
111            ProgrammableAddress::Subaddress1 => Register::SUBADDR1,
112            ProgrammableAddress::Subaddress2 => Register::SUBADDR2,
113            ProgrammableAddress::Subaddress3 => Register::SUBADDR3,
114            ProgrammableAddress::AllCall => Register::ALL_CALL_ADDR,
115        };
116        self.i2c
117            .write(self.address, &[reg, a.0])
118            .map_err(Error::I2C)
119    }

164    fn check_address(address: u8) -> Result<(), Error<E>> {
165        const LED_ALL_CALL: u8 = 0b111_0000;
166        // const SW_RESET: u8 = 0b000_0011; this gets absorbed by the high speed mode test
167        const HIGH_SPEED_MODE: u8 = 0b00_0111;
168        if address == 0 || address > 0x7F || address == LED_ALL_CALL || address <= HIGH_SPEED_MODE {
169            Err(Error::InvalidInputData)
170        } else {
171            Ok(())
172        }
173    }

175    /// Set the output change behavior. Either byte-by-byte or all at the same time.
176    ///
177    /// Note that update on ACK requires all 4 PWM channel registers to be loaded before
178    /// outputs are changed on the last ACK.
179    pub fn set_output_change_behavior(
180        &mut self,
181        change_behavior: OutputStateChange,
182    ) -> Result<(), Error<E>> {
183        let config = match change_behavior {
184            OutputStateChange::OnStop => self.config.with_low(BitFlagMode2::Och),
185            OutputStateChange::OnAck => self.config.with_high(BitFlagMode2::Och),
186        };
187        self.write_mode2(config)
188    }

190    /// Set the output driver configuration.
191    pub fn set_output_driver(&mut self, driver: OutputDriver) -> Result<(), Error<E>> {
192        let config = match driver {
193            OutputDriver::TotemPole => self.config.with_high(BitFlagMode2::OutDrv),
194            OutputDriver::OpenDrain => self.config.with_low(BitFlagMode2::OutDrv),

199    /// Set the output value when outputs are disabled (`OE` = 1).
200    pub fn set_disabled_output_value(
201        &mut self,
202        value: DisabledOutputValue,
203    ) -> Result<(), Error<E>> {
204        let config = match value {
205            DisabledOutputValue::Zero => self
206                .config
207                .with_low(BitFlagMode2::OutNe0)
208                .with_low(BitFlagMode2::OutNe1),
209            DisabledOutputValue::OutputDriver => self
210                .config
211                .with_high(BitFlagMode2::OutNe0)
212                .with_low(BitFlagMode2::OutNe1),
213            DisabledOutputValue::HighImpedance => self
214                .config
215                .with_low(BitFlagMode2::OutNe0)
216                .with_high(BitFlagMode2::OutNe1),
217        };
218        self.write_mode2(config)
219    }

243    /// Set the prescale value.
244    ///
245    /// The prescale value can be calculated for an update rate with the formula:
246    /// `prescale_value = round(osc_value / (4096 * update_rate)) - 1`
247    ///
248    /// The minimum prescale value is 3, which corresonds to an update rate of
249    /// 1526 Hz. The maximum prescale value is 255, which corresponds to an
250    /// update rate of 24 Hz.
251    ///
252    /// If you want to control a servo, set a prescale value of 100. This will
253    /// correspond to a frequency of about 60 Hz, which is the frequency at
254    /// which servos work.
255    ///
256    /// Internally this function stops the oscillator and restarts it after
257    /// setting the prescale value if it was running.
258    pub fn set_prescale(&mut self, prescale: u8) -> Result<(), Error<E>> {
259        if prescale < 3 {
260            return Err(Error::InvalidInputData);
261        }
262        let config = self.config;
263        let was_oscillator_running = config.is_low(BitFlagMode1::Sleep);
264        if was_oscillator_running {
265            // stop the oscillator
266            self.write_mode1(config.with_high(BitFlagMode1::Sleep))?;
267        }
268
269        self.i2c
270            .write(self.address, &[Register::PRE_SCALE, prescale])
271            .map_err(Error::I2C)?;
272
273        if was_oscillator_running {
274            // restart the oscillator
275            self.write_mode1(config)?;
276        }
277        Ok(())
278    }
