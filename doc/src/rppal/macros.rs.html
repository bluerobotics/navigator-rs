21macro_rules! parse_retval {
22    ($retval:expr) => {{
23        let retval = $retval;
24
25        if retval == -1 {
26            Err(std::io::Error::last_os_error())
27        } else {
28            Ok(retval)
29        }
30    }};
31}

33// Initialize an array with a type that doesn't implement Copy
34macro_rules! init_array {
35    ($val:expr, $len:expr) => {{
36        // Based on https://doc.rust-lang.org/stable/std/mem/union.MaybeUninit.html#initializing-an-array-element-by-element
37
38        use std::mem::{self, MaybeUninit};
39
40        // Create an uninitialized array of `MaybeUninit`. The `assume_init` is
41        // safe because the type we are claiming to have initialized here is a
42        // bunch of `MaybeUninit`s, which do not require initialization.
43        let mut data: [MaybeUninit<_>; $len] = unsafe { MaybeUninit::uninit().assume_init() };
44
45        // Dropping a `MaybeUninit` does nothing. Thus using raw pointer
46        // assignment instead of `ptr::write` does not cause the old
47        // uninitialized value to be dropped. Also if there is a panic during
48        // this loop, we have a memory leak, but there is no memory safety
49        // issue.
50        for elem in &mut data[..] {
51            *elem = MaybeUninit::new($val);
52        }
53
54        // Everything is initialized. Transmute the array to the
55        // initialized type.
56        unsafe { mem::transmute::<_, [_; $len]>(data) }
57    }};
58}
