21//! Interface for the PWM peripheral.
22//!
23//! RPPAL controls the Raspberry Pi's PWM peripheral through the `pwm` sysfs
24//! interface.
25//!
26//! ## PWM channels
27//!
28//! The BCM283x SoC supports two hardware PWM channels. By default, both channels
29//! are disabled. To enable only PWM0 on its default pin (BCM GPIO 18, physical pin 12),
30//! add `dtoverlay=pwm` to `/boot/config.txt` on Raspberry Pi OS or `boot/firmware/usercfg.txt` on
31//! Ubuntu. If you need both PWM channels, replace `pwm` with `pwm-2chan`, which enables PWM0 on BCM
32//! GPIO 18 (physical pin 12), and PWM1 on BCM GPIO 19 (physical pin 35). More details on enabling
33//! and configuring PWM on other GPIO pins than the default ones can be found in
34//! `/boot/overlays/README`.
35//!
36//! The Raspberry Pi's analog audio output uses both PWM channels. Playing audio and
37//! simultaneously accessing a PWM channel may cause issues.
38//!
39//! Some of the GPIO pins capable of supporting hardware PWM can also be configured for
40//! use with other peripherals. Be careful not to enable two peripherals on the same pin
41//! at the same time.
42//!
43//! ## Using PWM without superuser privileges (`sudo`)
44//!
45//! As of kernel version 4.14.34, released on April 16 2018, it's possible to
46//! configure your Raspberry Pi to allow non-root access to PWM. 4.14.34 includes
47//! a [patch] that allows `udev` to change file permissions when a
48//! PWM channel is exported. This will let any user that's a member of the `gpio`
49//! group configure PWM without having to use `sudo`.
50//!
51//! The `udev` rules needed to make this work haven't been patched in yet as of
52//! June 2018, but you can easily add them yourself. Make sure you're running
53//! 4.14.34 or later, and append the following snippet to
54//! `/etc/udev/rules.d/99-com.rules`. Reboot the Raspberry Pi afterwards.
55//!
56//! ```text
57//! SUBSYSTEM=="pwm*", PROGRAM="/bin/sh -c '\
58//!     chown -R root:gpio /sys/class/pwm && chmod -R 770 /sys/class/pwm;\
59//!     chown -R root:gpio /sys/devices/platform/soc/*.pwm/pwm/pwmchip* &&\
60//!     chmod -R 770 /sys/devices/platform/soc/*.pwm/pwm/pwmchip*\
61//! '"
62//! ```
63//!
64//! ## Troubleshooting
65//!
66//! ### Permission denied
67//!
68//! If [`new`] returns an `io::ErrorKind::PermissionDenied`
69//! error, make sure `/sys/class/pwm` and all of its subdirectories
70//! are owned by `root:gpio`, the current user is a member of the `gpio` group
71//! and `udev` is properly configured as mentioned above. Alternatively, you can
72//! launch your application using `sudo`.
73//!
74//! ### Not found
75//!
76//! If [`new`] returns an `io::ErrorKind::NotFound` error, you may have
77//! forgotten to enable the selected PWM channel. The configuration options
78//! to enable either of the two PWM channels are listed above.
79//!
80//! [patch]: https://github.com/raspberrypi/linux/issues/1983
81//! [`new`]: struct.Pwm.html#method.new

97/// Errors that can occur when accessing the PWM peripheral.
98#[derive(Debug)]
99pub enum Error {
100    /// I/O error.
101    Io(io::Error),
102}
103
104impl fmt::Display for Error {
105    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
106        match *self {
107            Error::Io(ref err) => write!(f, "I/O error: {}", err),
108        }
109    }
110}
111
112impl error::Error for Error {}
113
114impl From<io::Error> for Error {
115    fn from(err: io::Error) -> Error {
116        Error::Io(err)
117    }
118}

120/// Result type returned from methods that can have `pwm::Error`s.
121pub type Result<T> = result::Result<T, Error>;

160/// Provides access to the Raspberry Pi's PWM peripheral.
161///
162/// Before using `Pwm`, make sure the selected PWM channel has been configured
163/// and activated. More information can be found [here].
164///
165/// The `embedded-hal` [`PwmPin`] trait implementation for `Pwm` can be enabled
166/// by specifying the optional `hal` feature in the dependency declaration for
167/// the `rppal` crate.
168///
169/// The `unproven` `embedded-hal` [`Pwm`] trait implementation for `Pwm` can be enabled
170/// by specifying the optional `hal-unproven` feature in the dependency declaration for
171/// the `rppal` crate.
172///
173/// [here]: index.html
174/// [`PwmPin`]: ../../embedded_hal/trait.PwmPin.html
175/// [`Pwm`]: ../../embedded_hal/trait.Pwm.html
176#[derive(Debug)]
177pub struct Pwm {
178    channel: Channel,
179    reset_on_drop: bool,
180}
181
182impl Pwm {
183    /// Constructs a new `Pwm`.
184    ///
185    /// `new` doesn't change the channel's period, pulse width or polarity. The channel
186    /// will remain disabled until [`enable`] is called.
187    ///
188    /// [`enable`]: #method.enable
189    pub fn new(channel: Channel) -> Result<Pwm> {
190        sysfs::export(channel as u8)?;
191
192        let pwm = Pwm {
193            channel,
194            reset_on_drop: true,
195        };
196
197        // Always reset "enable" to 0. The sysfs interface has a bug where a previous
198        // export may have left "enable" as 1 after unexporting. On the next export,
199        // "enable" is still set to 1, even though the channel isn't enabled.
200        let _ = pwm.disable();
201
202        Ok(pwm)
203    }
204
205    /// Constructs a new `Pwm` using the specified settings.
206    ///
207    /// `period` indicates the time it takes for the PWM channel to complete one cycle.
208    ///
209    /// `pulse_width` indicates the amount of time the PWM channel is active during a
210    /// single period.
211    ///
212    /// `polarity` configures the active logic level as either high ([`Normal`])
213    /// or low ([`Inverse`]).
214    ///
215    /// `enabled` enables PWM on the selected channel. If `enabled` is set to `false`,
216    /// the channel will remain disabled until [`enable`] is called.
217    ///
218    /// This method will fail if `period` is shorter than `pulse_width`.
219    ///
220    /// [`Normal`]: enum.Polarity.html#variant.Normal
221    /// [`Inverse`]: enum.Polarity.html#variant.Inverse
222    /// [`enable`]: #method.enable
223    pub fn with_period(
224        channel: Channel,
225        period: Duration,
226        pulse_width: Duration,
227        polarity: Polarity,
228        enabled: bool,
229    ) -> Result<Pwm> {
230        sysfs::export(channel as u8)?;
231
232        let pwm = Pwm {
233            channel,
234            reset_on_drop: true,
235        };
236
237        // Always reset "enable" to 0. The sysfs pwm interface has a bug where a previous
238        // export may have left "enable" as 1 after unexporting. On the next export,
239        // "enable" is still set to 1, even though the channel isn't enabled.
240        let _ = pwm.disable();
241
242        // Set pulse width to 0 first in case the new period is shorter than the current pulse width
243        let _ = sysfs::set_pulse_width(channel as u8, 0);
244
245        pwm.set_period(period)?;
246        pwm.set_pulse_width(pulse_width)?;
247        pwm.set_polarity(polarity)?;
248        if enabled {
249            pwm.enable()?;
250        }
251
252        Ok(pwm)
253    }
254
255    /// Constructs a new `Pwm` using the specified settings.
256    ///
257    /// `with_frequency` is a convenience method that converts `frequency` to a period,
258    /// and calculates the duty cycle as a percentage of the frequency.
259    ///
260    /// `frequency` is specified in hertz (Hz).
261    ///
262    /// `duty_cycle` is specified as a floating point value between `0.0` (0%) and `1.0` (100%).
263    ///
264    /// `polarity` configures the active logic level as either high ([`Normal`])
265    /// or low ([`Inverse`]).
266    ///
267    /// `enabled` enables PWM on the selected channel. If `enabled` is set to `false`,
268    /// the channel will remain disabled until [`enable`] is called.
269    ///
270    /// [`Normal`]: enum.Polarity.html#variant.Normal
271    /// [`Inverse`]: enum.Polarity.html#variant.Inverse
272    /// [`enable`]: #method.enable
273    pub fn with_frequency(
274        channel: Channel,
275        frequency: f64,
276        duty_cycle: f64,
277        polarity: Polarity,
278        enabled: bool,
279    ) -> Result<Pwm> {
280        sysfs::export(channel as u8)?;
281
282        let pwm = Pwm {
283            channel,
284            reset_on_drop: true,
285        };
286
287        // Always reset "enable" to 0. The sysfs pwm interface has a bug where a previous
288        // export may have left "enable" as 1 after unexporting. On the next export,
289        // "enable" is still set to 1, even though the channel isn't enabled.
290        let _ = pwm.disable();
291
292        // Set pulse width to 0 first in case the new period is shorter than the current pulse width
293        let _ = sysfs::set_pulse_width(channel as u8, 0);
294
295        // Convert to nanoseconds
296        let period = if frequency == 0.0 {
297            0.0
298        } else {
299            (1.0 / frequency) * NANOS_PER_SEC
300        };
301        let pulse_width = period * duty_cycle.max(0.0).min(1.0);
302
303        sysfs::set_period(channel as u8, period as u64)?;
304        sysfs::set_pulse_width(channel as u8, pulse_width as u64)?;
305        pwm.set_polarity(polarity)?;
306        if enabled {
307            pwm.enable()?;
308        }
309
310        Ok(pwm)
311    }

313    /// Returns the period.
314    pub fn period(&self) -> Result<Duration> {
315        Ok(Duration::from_nanos(sysfs::period(self.channel as u8)?))
316    }
317
318    /// Sets the period.
319    ///
320    /// `period` indicates the time it takes for the PWM channel to complete one cycle.
321    ///
322    /// This method will fail if `period` is shorter than the current pulse width.
323    pub fn set_period(&self, period: Duration) -> Result<()> {
324        sysfs::set_period(
325            self.channel as u8,
326            u64::from(period.subsec_nanos())
327                .saturating_add(period.as_secs().saturating_mul(NANOS_PER_SEC as u64)),
328        )?;
329
330        Ok(())
331    }
332
333    /// Returns the pulse width.
334    pub fn pulse_width(&self) -> Result<Duration> {
335        Ok(Duration::from_nanos(sysfs::pulse_width(
336            self.channel as u8,
337        )?))
338    }
339
340    /// Sets the pulse width.
341    ///
342    /// `pulse_width` indicates the amount of time the PWM channel is active during a
343    /// single period.
344    ///
345    /// This method will fail if `pulse_width` is longer than the current period.
346    pub fn set_pulse_width(&self, pulse_width: Duration) -> Result<()> {
347        sysfs::set_pulse_width(
348            self.channel as u8,
349            u64::from(pulse_width.subsec_nanos())
350                .saturating_add(pulse_width.as_secs().saturating_mul(NANOS_PER_SEC as u64)),
351        )?;
352
353        Ok(())
354    }
355
356    /// Returns the frequency.
357    ///
358    /// `frequency` is a convenience method that calculates the frequency in hertz (Hz)
359    /// based on the configured period.
360    pub fn frequency(&self) -> Result<f64> {
361        let period = sysfs::period(self.channel as u8)? as f64;
362
363        Ok(if period == 0.0 {
364            0.0
365        } else {
366            1.0 / (period / NANOS_PER_SEC)
367        })
368    }
369
370    /// Sets the frequency and duty cycle.
371    ///
372    /// `set_frequency` is a convenience method that converts `frequency` to a period,
373    /// and calculates the duty cycle as a percentage of the frequency.
374    ///
375    /// `frequency` is specified in hertz (Hz).
376    ///
377    /// `duty_cycle` is specified as a floating point value between `0.0` (0%) and `1.0` (100%).
378    pub fn set_frequency(&self, frequency: f64, duty_cycle: f64) -> Result<()> {
379        // Set duty cycle to 0 first in case the new period is shorter than the current duty cycle
380        let _ = sysfs::set_pulse_width(self.channel as u8, 0);
381
382        // Convert to nanoseconds
383        let period = if frequency == 0.0 {
384            0.0
385        } else {
386            (1.0 / frequency) * NANOS_PER_SEC
387        };
388        let pulse_width = period * duty_cycle.max(0.0).min(1.0);
389
390        sysfs::set_period(self.channel as u8, period as u64)?;
391        sysfs::set_pulse_width(self.channel as u8, pulse_width as u64)?;
392
393        Ok(())
394    }
395
396    /// Returns the duty cycle.
397    ///
398    /// `duty_cycle` is a convenience method that calculates the duty cycle as a
399    /// floating point value between `0.0` (0%) and `1.0` (100%) based on the configured
400    /// period and pulse width.
401    pub fn duty_cycle(&self) -> Result<f64> {
402        let period = sysfs::period(self.channel as u8)? as f64;
403        let pulse_width = sysfs::pulse_width(self.channel as u8)? as f64;
404
405        Ok(if period == 0.0 {
406            0.0
407        } else {
408            (pulse_width / period).max(0.0).min(1.0)
409        })
410    }
411
412    /// Sets the duty cycle.
413    ///
414    /// `set_duty_cycle` is a convenience method that converts `duty_cycle` to a
415    /// pulse width based on the configured period.
416    ///
417    /// `duty_cycle` is specified as a floating point value between `0.0` (0%) and `1.0` (100%).
418    pub fn set_duty_cycle(&self, duty_cycle: f64) -> Result<()> {
419        let period = sysfs::period(self.channel as u8)? as f64;
420        let pulse_width = period * duty_cycle.max(0.0).min(1.0);
421
422        sysfs::set_pulse_width(self.channel as u8, pulse_width as u64)?;
423
424        Ok(())
425    }

469    /// When enabled, disables the PWM channel when the `Pwm` instance
470    /// goes out of scope. By default, this is set to `true`.
471    ///
472    /// ## Note
473    ///
474    /// Drop methods aren't called when a process is abnormally terminated, for
475    /// instance when a user presses <kbd>Ctrl</kbd> + <kbd>C</kbd>, and the `SIGINT` signal
476    /// isn't caught. You can catch those using crates such as [`simple_signal`].
477    ///
478    /// [`simple_signal`]: https://crates.io/crates/simple-signal
479    pub fn set_reset_on_drop(&mut self, reset_on_drop: bool) {
480        self.reset_on_drop = reset_on_drop;
481    }
482}
483
484impl Drop for Pwm {
485    fn drop(&mut self) {
486        if self.reset_on_drop {
487            let _ = sysfs::set_enabled(self.channel as u8, false);
488            let _ = sysfs::unexport(self.channel as u8);
489        }
490    }
