34const PERIPHERAL_BASE_RPI: u32 = 0x2000_0000;
35const PERIPHERAL_BASE_RPI2: u32 = 0x3f00_0000;

39/// Errors that can occur when trying to identify the Raspberry Pi hardware.
40#[derive(Debug)]
41pub enum Error {
42    /// Unknown model.
43    ///
44    /// `DeviceInfo` was unable to identify the Raspberry Pi model based on the
45    /// contents of `/proc/cpuinfo`, `/sys/firmware/devicetree/base/compatible`
46    /// and `/sys/firmware/devicetree/base/model`.
47    ///
48    /// Support for new models is usually added shortly after they are officially
49    /// announced and available to the public. Make sure you're using the latest
50    /// release of RPPAL.
51    ///
52    /// You may also encounter this error if your Linux distribution
53    /// doesn't provide any of the common user-accessible system files
54    /// that are used to identify the model and SoC.
55    UnknownModel,
56}
57
58impl fmt::Display for Error {
59    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
60        match *self {
61            Error::UnknownModel => write!(f, "Unknown Raspberry Pi model"),
62        }
63    }
64}
65
66impl error::Error for Error {}

124/// Identifiable Raspberry Pi SoCs.

130#[non_exhaustive]

151// Identify Pi model based on /proc/cpuinfo
152fn parse_proc_cpuinfo() -> Result<Model> {
153    let proc_cpuinfo = BufReader::new(match File::open("/proc/cpuinfo") {
154        Ok(file) => file,
155        Err(_) => return Err(Error::UnknownModel),
156    });
157
158    let mut hardware: String = String::new();
159    let mut revision: String = String::new();
160    for line in proc_cpuinfo.lines().flatten() {
161        if let Some(line_value) = line.strip_prefix("Hardware\t: ") {
162            hardware = String::from(line_value);
163        } else if let Some(line_value) = line.strip_prefix("Revision\t: ") {
164            revision = String::from(line_value).to_lowercase();
165        }
166    }
167
168    // Return an error if we don't recognize the SoC. This check is
169    // done to prevent accidentally identifying a non-Pi SBC as a Pi
170    // solely based on the revision field.
171    match &hardware[..] {
172        "BCM2708" | "BCM2835" | "BCM2709" | "BCM2836" | "BCM2710" | "BCM2837" | "BCM2837A1"
173        | "BCM2837B0" | "RP3A0-AU" | "BCM2710A1" | "BCM2711" => {}
174        _ => return Err(Error::UnknownModel),
175    }
176
177    let model = if (revision.len() == 4) || (revision.len() == 8) {
178        // Older revisions are 4 characters long, or 8 if they've been over-volted
179        match &revision[revision.len() - 4..] {
180            "0007" | "0008" | "0009" | "0015" => Model::RaspberryPiA,
181            "beta" | "0002" | "0003" => Model::RaspberryPiBRev1,
182            "0004" | "0005" | "0006" | "000d" | "000e" | "000f" => Model::RaspberryPiBRev2,
183            "0012" => Model::RaspberryPiAPlus,
184            "0010" | "0013" => Model::RaspberryPiBPlus,
185            "0011" | "0014" => Model::RaspberryPiComputeModule,
186            _ => return Err(Error::UnknownModel),
187        }
188    } else if revision.len() >= 6 {
189        // Newer revisions consist of at least 6 characters
190        match &revision[..] {
191            "900021" => Model::RaspberryPiAPlus,
192            "900032" => Model::RaspberryPiBPlus,
193            "a01040" | "a01041" | "a21041" | "a22042" => Model::RaspberryPi2B,
194            "a02082" | "a22082" | "a32082" | "a52082" => Model::RaspberryPi3B,
195            "900092" | "900093" | "920092" | "920093" => Model::RaspberryPiZero,
196            "a020a0" | "a220a0" => Model::RaspberryPiComputeModule3,
197            "9000c1" => Model::RaspberryPiZeroW,
198            "a020d3" => Model::RaspberryPi3BPlus,
199            "9020e0" => Model::RaspberryPi3APlus,
200            "a02100" => Model::RaspberryPiComputeModule3Plus,
201            "a03111" | "a03112" | "b03111" | "b03112" | "b03114" | "c03111" | "c03112"
202            | "c03114" | "d03114" => Model::RaspberryPi4B,
203            "c03130" => Model::RaspberryPi400,
204            "a03140" | "b03140" | "c03140" | "d03140" => Model::RaspberryPiComputeModule4,
205            "902120" => Model::RaspberryPiZero2W,
206            _ => return Err(Error::UnknownModel),
207        }
208    } else {
209        return Err(Error::UnknownModel);
210    };
211
212    Ok(model)
213}

215// Identify Pi model based on /sys/firmware/devicetree/base/compatible
216fn parse_base_compatible() -> Result<Model> {
217    let base_compatible = match fs::read_to_string("/sys/firmware/devicetree/base/compatible") {
218        Ok(buffer) => buffer,
219        Err(_) => return Err(Error::UnknownModel),
220    };
221
222    // Based on /arch/arm/boot/dts/ and /Documentation/devicetree/bindings/arm/bcm/
223    for comp_id in base_compatible.split('\0') {
224        let model = match comp_id {
225            "raspberrypi,model-b-i2c0" => Model::RaspberryPiBRev1,
226            "raspberrypi,model-b" => Model::RaspberryPiBRev1,
227            "raspberrypi,model-a" => Model::RaspberryPiA,
228            "raspberrypi,model-b-rev2" => Model::RaspberryPiBRev2,
229            "raspberrypi,model-a-plus" => Model::RaspberryPiAPlus,
230            "raspberrypi,model-b-plus" => Model::RaspberryPiBPlus,
231            "raspberrypi,2-model-b" => Model::RaspberryPi2B,
232            "raspberrypi,compute-module" => Model::RaspberryPiComputeModule,
233            "raspberrypi,3-model-b" => Model::RaspberryPi3B,
234            "raspberrypi,model-zero" => Model::RaspberryPiZero,
235            "raspberrypi,3-compute-module" => Model::RaspberryPiComputeModule3,
236            "raspberrypi,3-compute-module-plus" => Model::RaspberryPiComputeModule3Plus,
237            "raspberrypi,model-zero-w" => Model::RaspberryPiZeroW,
238            "raspberrypi,model-zero-2" => Model::RaspberryPiZero2W,
239            "raspberrypi,3-model-b-plus" => Model::RaspberryPi3BPlus,
240            "raspberrypi,3-model-a-plus" => Model::RaspberryPi3APlus,
241            "raspberrypi,4-model-b" => Model::RaspberryPi4B,
242            "raspberrypi,400" => Model::RaspberryPi400,
243            "raspberrypi,4-compute-module" => Model::RaspberryPiComputeModule4,
244            _ => continue,
245        };
246
247        return Ok(model);
248    }
249
250    Err(Error::UnknownModel)
251}

253// Identify Pi model based on /sys/firmware/devicetree/base/model
254fn parse_base_model() -> Result<Model> {
255    let mut base_model = match fs::read_to_string("/sys/firmware/devicetree/base/model") {
256        Ok(mut buffer) => {
257            if let Some(idx) = buffer.find('\0') {
258                buffer.truncate(idx);
259            }
260
261            buffer
262        }
263        Err(_) => return Err(Error::UnknownModel),
264    };
265
266    // Check if this is a Pi B rev 2 before we remove the revision part, assuming the
267    // PCB Revision numbers on https://elinux.org/RPi_HardwareHistory are correct, and
268    // the installed distro appends the revision to the model name.
269    match &base_model[..] {
270        "Raspberry Pi Model B Rev 2.0" => return Ok(Model::RaspberryPiBRev2),
271        "Raspberry Pi Model B rev2 Rev 2.0" => return Ok(Model::RaspberryPiBRev2),
272        _ => (),
273    }
274
275    if let Some(idx) = base_model.find(" Rev ") {
276        base_model.truncate(idx);
277    }
278
279    // Based on /arch/arm/boot/dts/ and /Documentation/devicetree/bindings/arm/bcm/
280    let model = match &base_model[..] {
281        "Raspberry Pi Model B (no P5)" => Model::RaspberryPiBRev1,
282        "Raspberry Pi Model B" => Model::RaspberryPiBRev1,
283        "Raspberry Pi Model A" => Model::RaspberryPiA,
284        "Raspberry Pi Model B rev2" => Model::RaspberryPiBRev2,
285        "Raspberry Pi Model A+" => Model::RaspberryPiAPlus,
286        "Raspberry Pi Model A Plus" => Model::RaspberryPiAPlus,
287        "Raspberry Pi Model B+" => Model::RaspberryPiBPlus,
288        "Raspberry Pi Model B Plus" => Model::RaspberryPiBPlus,
289        "Raspberry Pi 2 Model B" => Model::RaspberryPi2B,
290        "Raspberry Pi Compute Module" => Model::RaspberryPiComputeModule,
291        "Raspberry Pi 3 Model B" => Model::RaspberryPi3B,
292        "Raspberry Pi Zero" => Model::RaspberryPiZero,
293        "Raspberry Pi Compute Module 3" => Model::RaspberryPiComputeModule3,
294        "Raspberry Pi Compute Module 3 Plus" => Model::RaspberryPiComputeModule3Plus,
295        "Raspberry Pi Zero W" => Model::RaspberryPiZeroW,
296        "Raspberry Pi Zero 2" => Model::RaspberryPiZero2W,
297        "Raspberry Pi 3 Model B+" => Model::RaspberryPi3BPlus,
298        "Raspberry Pi 3 Model B Plus" => Model::RaspberryPi3BPlus,
299        "Raspberry Pi 3 Model A Plus" => Model::RaspberryPi3APlus,
300        "Raspberry Pi 4 Model B" => Model::RaspberryPi4B,
301        "Raspberry Pi 400" => Model::RaspberryPi400,
302        "Raspberry Pi Compute Module 4" => Model::RaspberryPiComputeModule4,
303        _ => return Err(Error::UnknownModel),
304    };
305
306    Ok(model)
307}

309/// Retrieves Raspberry Pi device information.
310#[derive(Debug, PartialEq, Eq, Copy, Clone)]
311pub struct DeviceInfo {
312    model: Model,
313    soc: SoC,
314    peripheral_base: u32,
315    gpio_offset: u32,
316}
317
318impl DeviceInfo {
319    /// Constructs a new `DeviceInfo`.
320    ///
321    /// `new` attempts to identify the Raspberry Pi's model and SoC based on
322    /// the contents of `/proc/cpuinfo`, `/sys/firmware/devicetree/base/compatible`
323    /// and `/sys/firmware/devicetree/base/model`.
324    pub fn new() -> Result<DeviceInfo> {
325        // Parse order from most-detailed to least-detailed info
326        let model = parse_proc_cpuinfo()
327            .or_else(|_| parse_base_compatible().or_else(|_| parse_base_model()))?;
328
329        // Set SoC and memory offsets based on model
330        match model {
331            Model::RaspberryPiA
332            | Model::RaspberryPiAPlus
333            | Model::RaspberryPiBRev1
334            | Model::RaspberryPiBRev2
335            | Model::RaspberryPiBPlus
336            | Model::RaspberryPiComputeModule
337            | Model::RaspberryPiZero
338            | Model::RaspberryPiZeroW => Ok(DeviceInfo {
339                model,
340                soc: SoC::Bcm2835,
341                peripheral_base: PERIPHERAL_BASE_RPI,
342                gpio_offset: GPIO_OFFSET,
343            }),
344            Model::RaspberryPi2B => Ok(DeviceInfo {
345                model,
346                soc: SoC::Bcm2836,
347                peripheral_base: PERIPHERAL_BASE_RPI2,
348                gpio_offset: GPIO_OFFSET,
349            }),
350            Model::RaspberryPi3B | Model::RaspberryPiComputeModule3 | Model::RaspberryPiZero2W => {
351                Ok(DeviceInfo {
352                    model,
353                    soc: SoC::Bcm2837A1,
354                    peripheral_base: PERIPHERAL_BASE_RPI2,
355                    gpio_offset: GPIO_OFFSET,
356                })
357            }
358            Model::RaspberryPi3BPlus
359            | Model::RaspberryPi3APlus
360            | Model::RaspberryPiComputeModule3Plus => Ok(DeviceInfo {
361                model,
362                soc: SoC::Bcm2837B0,
363                peripheral_base: PERIPHERAL_BASE_RPI2,
364                gpio_offset: GPIO_OFFSET,
365            }),
366            Model::RaspberryPi4B | Model::RaspberryPi400 | Model::RaspberryPiComputeModule4 => {
367                Ok(DeviceInfo {
368                    model,
369                    soc: SoC::Bcm2711,
370                    peripheral_base: PERIPHERAL_BASE_RPI4,
371                    gpio_offset: GPIO_OFFSET,
372                })
373            }
374        }
375    }
376
377    /// Returns the Raspberry Pi's model.
378    pub fn model(&self) -> Model {
379        self.model
380    }
381
382    /// Returns the Raspberry Pi's SoC.
383    pub fn soc(&self) -> SoC {
384        self.soc
385    }
386
387    /// Returns the base memory address for the BCM283x peripherals.
388    pub(crate) fn peripheral_base(&self) -> u32 {
389        self.peripheral_base
390    }
391
392    /// Returns the offset from the base memory address for the GPIO section.
393    pub(crate) fn gpio_offset(&self) -> u32 {
394        self.gpio_offset
395    }
