37// Only call sleep_ns() if we have enough time remaining
38const SLEEP_THRESHOLD: i64 = 250_000;
39// Reserve some time for busy waiting
40const BUSYWAIT_MAX: i64 = 200_000;
41// Subtract from the remaining busy wait time to account for get_time_ns() overhead
42const BUSYWAIT_REMAINDER: i64 = 100;
43
44const NANOS_PER_SEC: i64 = 1_000_000_000;
45
46#[derive(Debug, PartialEq, Eq, Copy, Clone)]
47enum Msg {
48    Reconfigure(Duration, Duration),
49    Stop,
50}
51
52#[derive(Debug)]
53pub(crate) struct SoftPwm {
54    pwm_thread: Option<thread::JoinHandle<Result<()>>>,
55    sender: Sender<Msg>,
56}
57
58impl SoftPwm {
59    pub(crate) fn new(
60        pin: u8,
61        gpio_state: Arc<GpioState>,
62        period: Duration,
63        pulse_width: Duration,
64    ) -> SoftPwm {
65        let (sender, receiver): (Sender<Msg>, Receiver<Msg>) = mpsc::channel();
66
67        let pwm_thread = thread::spawn(move || -> Result<()> {
68            // Set the scheduling policy to real-time round robin at the highest priority. This
69            // will silently fail if we're not running as root.
70            #[cfg(target_env = "gnu")]
71            let params = sched_param {
72                sched_priority: unsafe { libc::sched_get_priority_max(SCHED_RR) },
73            };
74
75            #[cfg(target_env = "musl")]
76            let params = sched_param {
77                sched_priority: unsafe { libc::sched_get_priority_max(SCHED_RR) },
78                sched_ss_low_priority: 0,
79                sched_ss_repl_period: timespec {
80                    tv_sec: 0,
81                    tv_nsec: 0,
82                },
83                sched_ss_init_budget: timespec {
84                    tv_sec: 0,
85                    tv_nsec: 0,
86                },
87                sched_ss_max_repl: 0,
88            };
89
90            unsafe {
91                libc::sched_setscheduler(0, SCHED_RR, &params);
92            }
93
94            // Set timer slack to 1 ns (default = 50 µs). This is only relevant if we're unable
95            // to set a real-time scheduling policy.
96            unsafe {
97                libc::prctl(PR_SET_TIMERSLACK, 1);
98            }
99
100            let mut period_ns = period.as_nanos() as i64;
101            let mut pulse_width_ns = pulse_width.as_nanos() as i64;
102
103            let mut start_ns = get_time_ns();
104
105            loop {
106                // PWM active
107                if pulse_width_ns > 0 {
108                    gpio_state.gpio_mem.set_high(pin);
109                }
110
111                // Sleep if we have enough time remaining, while reserving some time
112                // for busy waiting to compensate for sleep taking longer than needed.
113                if pulse_width_ns >= SLEEP_THRESHOLD {
114                    sleep_ns(pulse_width_ns - BUSYWAIT_MAX);
115                }
116
117                // Busy-wait for the remaining active time, minus BUSYWAIT_REMAINDER
118                // to account for get_time_ns() overhead
119                loop {
120                    if (pulse_width_ns - (get_time_ns() - start_ns)) <= BUSYWAIT_REMAINDER {
121                        break;
122                    }
123                }
124
125                // PWM inactive
126                gpio_state.gpio_mem.set_low(pin);
127
128                while let Ok(msg) = receiver.try_recv() {
129                    match msg {
130                        Msg::Reconfigure(period, pulse_width) => {
131                            // Reconfigure period and pulse width
132                            pulse_width_ns = pulse_width.as_nanos() as i64;
133                            period_ns = period.as_nanos() as i64;
134
135                            if pulse_width_ns > period_ns {
136                                pulse_width_ns = period_ns;
137                            }
138                        }
139                        Msg::Stop => {
140                            // The main thread asked us to stop
141                            return Ok(());
142                        }
143                    }
144                }
145
146                let remaining_ns = period_ns - (get_time_ns() - start_ns);
147
148                // Sleep if we have enough time remaining, while reserving some time
149                // for busy waiting to compensate for sleep taking longer than needed.
150                if remaining_ns >= SLEEP_THRESHOLD {
151                    sleep_ns(remaining_ns - BUSYWAIT_MAX);
152                }
153
154                // Busy-wait for the remaining inactive time, minus BUSYWAIT_REMAINDER
155                // to account for get_time_ns() overhead
156                loop {
157                    let current_ns = get_time_ns();
158                    if (period_ns - (current_ns - start_ns)) <= BUSYWAIT_REMAINDER {
159                        start_ns = current_ns;
160                        break;
161                    }
162                }
163            }
164        });
165
166        SoftPwm {
167            pwm_thread: Some(pwm_thread),
168            sender,
169        }
170    }
171
172    pub(crate) fn reconfigure(&mut self, period: Duration, pulse_width: Duration) {
173        let _ = self.sender.send(Msg::Reconfigure(period, pulse_width));
174    }
175
176    pub(crate) fn stop(&mut self) -> Result<()> {
177        let _ = self.sender.send(Msg::Stop);
178        if let Some(pwm_thread) = self.pwm_thread.take() {
179            match pwm_thread.join() {
180                Ok(r) => return r,
181                Err(_) => return Err(Error::ThreadPanic),
182            }
183        }
184
185        Ok(())
186    }
187}
188
189impl Drop for SoftPwm {
190    fn drop(&mut self) {
191        // Don't wait for the pwm thread to exit if the main thread is panicking,
192        // because we could potentially block indefinitely while unwinding if the
193        // pwm thread doesn't respond to the Stop message for some reason.
194        if !thread::panicking() {
195            let _ = self.stop();
196        }
197    }
198}

200// Required because Sender isn't Sync. Implementing Sync for SoftPwm is
201// safe because all usage of Sender::send() is locked behind &mut self.
202unsafe impl Sync for SoftPwm {}

204#[inline(always)]
205fn get_time_ns() -> i64 {
206    let mut ts = timespec {
207        tv_sec: 0,
208        tv_nsec: 0,
209    };
210
211    unsafe {
212        libc::clock_gettime(CLOCK_MONOTONIC, &mut ts);
213    }
214
215    (ts.tv_sec as i64 * NANOS_PER_SEC) + ts.tv_nsec as i64
216}
217
218#[inline(always)]
219fn sleep_ns(ns: i64) {
220    let ts = timespec {
221        tv_sec: (ns / NANOS_PER_SEC) as time_t,
222        tv_nsec: (ns % NANOS_PER_SEC) as c_long,
223    };
224
225    unsafe {
226        libc::clock_nanosleep(CLOCK_MONOTONIC, 0, &ts, ptr::null_mut());
227    }
228}
