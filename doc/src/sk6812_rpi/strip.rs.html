12const SPI_FREQUENCY: u32 = 6_400_000;
13
14/// Structure representing whole SK6812RGBW strip.
15/// Should be compatible with other similar LED's, but they would likely require a different bit ordering
16#[derive(Debug)]
17pub struct Strip {
18    spi: Spi,
19    pub leds: Vec<Led>,
20}
21
22impl Strip {
23    /// Create new SK6812RGBW strip
24    /// Since rppal library requires slave-select pin to initalize SPI, by default SS0 is selected. It's not used to drive LEDs, so it's a wasted pin.
25    /// If you want to select other pin, use `new_with_custom_ss` method.
26    pub fn new(bus: Bus, amount_of_leds: usize) -> Result<Self, Box<dyn Error>> {
27        Ok(Self {
28            spi: Spi::new(bus, SlaveSelect::Ss0, SPI_FREQUENCY, Mode::Mode0)?,
29            leds: vec![Led::new(); amount_of_leds],
30        })
31    }
32
33    /// Create new SK6812RGBW strip with custom slave-select pin
34    /// If you want to use SS0 for different purposes, you can waste another pin with this function instead.
35    pub fn new_with_custom_ss(
36        bus: Bus,
37        amount_of_leds: usize,
38        slave_select: SlaveSelect,
39    ) -> Result<Self, Box<dyn Error>> {
40        Ok(Self {
41            spi: Spi::new(bus, slave_select, SPI_FREQUENCY, Mode::Mode0)?,
42            leds: vec![Led::new(); amount_of_leds],
43        })
44    }
45
46    /// Set the color of all LEDs in the strip at once
47    pub fn fill(&mut self, led: Led) {
48        self.leds.fill(led);
49    }
50
51    // Turn off all the LEDs
52    pub fn clear(&mut self) {
53        self.leds.fill(Led::new());
54    }
55
56    // Set the strip LED colors to a `palette` gradient
57    pub fn set_gradient(&mut self, gradient: Gradient<LinSrgb>) {
58        gradient
59            .take(self.leds.len())
60            .zip(&mut self.leds)
61            .for_each(|(color, led)| {
62                *led = Srgb::from_linear(color).into();
63            });
64    }
65
66    // "Rotate" LEDs to the beginning of the strip - move the first LED color to the end, 2nd to the first, and so on.
67    // This description assumes that strip goes left-to-right
68    pub fn shift_left(&mut self, count: usize) {
69        self.leds.rotate_left(count);
70    }
71
72    // "Rotate" LEDs to the end of the strip - move the last LED color to the beginning, 1st to 2nd, and so on.
73    // This description assumes that strip goes left-to-right
74    pub fn shift_right(&mut self, count: usize) {
75        self.leds.rotate_right(count);
76    }
77
78    /// Call this to send the data from `leds` to the strip
79    /// This function will block the thread for ~80us after sending the data,
80    /// which is caused by strip comms protocol requirements.
81    ///
82    /// If you're getting an error, telling you that the message is too long - increase the SPI transfer size in `/boot/cmdline.txt`.
83    /// To do so, add `spidev.bufsiz=65535` to the first line of this file. I added it right before `rootwait`, but placement shouldn't matter.
84    pub fn update(&mut self) -> Result<(), Box<dyn Error>> {
85        let led_data: Vec<u8> = self.raw_led_data().collect();
86        self.spi.write(&led_data)?;
87        thread::sleep(Duration::from_micros(80));
88
89        Ok(())
90    }
91
92    // Get raw LED bytes, ready to send via SPI. This function is called internally by the library and usually you don't need to worry about it.
93    fn raw_led_data(&self) -> impl Iterator<Item = u8> + '_ {
94        self.leds.iter().flat_map(|led| led.to_raw_led_bytes())
95    }
96}
97
98impl ShrAssign<usize> for Strip {
99    fn shr_assign(&mut self, rhs: usize) {
100        self.shift_right(rhs);
101    }
102}
103
104impl ShlAssign<usize> for Strip {
105    fn shl_assign(&mut self, rhs: usize) {
106        self.shift_left(rhs);
107    }
108}

110#[cfg(test)]
111mod tests {
112    use super::*;
113
114    fn make_strip() -> Strip {
115        Strip::new(Bus::Spi0, 144).unwrap()
116    }
117
118    #[test]
119    fn test_setting_color() {
120        let led: Led = [100, 0, 0].into();
121        let mut strip = make_strip();
122
123        strip.fill(led);
124
125        strip.leds.iter().for_each(|strip_led| {
126            assert_eq!(*strip_led, led);
127        })
128    }
129
130    #[test]
131    fn test_clearing() {
132        let led: Led = [100, 0, 0].into();
133        let mut strip = make_strip();
134
135        strip.fill(led);
136        strip.clear();
137
138        strip.leds.iter().for_each(|strip_led| {
139            assert_eq!(*strip_led, Led::new());
140        })
141    }
142
143    #[test]
144    fn test_shift_right() {
145        let mut strip = Strip::new(Bus::Spi0, 5).unwrap();
146
147        strip.leds[0].r = 1;
148        strip.leds[1].r = 2;
149        strip.leds[2].r = 3;
150        strip.leds[3].r = 4;
151        strip.leds[4].r = 5;
152
153        strip >>= 1;
154
155        assert_eq!(strip.leds[0].r, 5);
156        assert_eq!(strip.leds[1].r, 1);
157        assert_eq!(strip.leds[2].r, 2);
158        assert_eq!(strip.leds[3].r, 3);
159        assert_eq!(strip.leds[4].r, 4);
160    }
161
162    #[test]
163    fn test_shift_left() {
164        let mut strip = Strip::new(Bus::Spi0, 5).unwrap();
165
166        strip.leds[0].r = 1;
167        strip.leds[1].r = 2;
168        strip.leds[2].r = 3;
169        strip.leds[3].r = 4;
170        strip.leds[4].r = 5;
171
172        strip <<= 1;
173
174        assert_eq!(strip.leds[0].r, 2);
175        assert_eq!(strip.leds[1].r, 3);
176        assert_eq!(strip.leds[2].r, 4);
177        assert_eq!(strip.leds[3].r, 5);
178        assert_eq!(strip.leds[4].r, 1);
179    }
