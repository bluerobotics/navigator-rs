2
3use crate::{
4    ic, interface::I2cInterface, mode, Ads1x1x, Config, FullScaleRange, SlaveAddr,
5    DEVICE_BASE_ADDRESS,
6};
7use core::marker::PhantomData;
8use embedded_hal::blocking;
9
10macro_rules! impl_new_destroy {
11    ( $IC:ident, $create:ident, $destroy:ident, $conv:ty ) => {
12        impl<I2C, E> Ads1x1x<I2cInterface<I2C>, ic::$IC, $conv, mode::OneShot>
13        where
14            I2C: blocking::i2c::Write<Error = E> + blocking::i2c::WriteRead<Error = E>,
15        {
16            /// Create a new instance of the device in OneShot mode.
17            pub fn $create(i2c: I2C, address: SlaveAddr) -> Self {
18                Ads1x1x {
19                    iface: I2cInterface {
20                        i2c,
21                        address: address.addr(DEVICE_BASE_ADDRESS),
22                    },
23                    config: Config::default(),
24                    fsr: FullScaleRange::default(),
25                    a_conversion_was_started: false,
26                    _conv: PhantomData,
27                    _ic: PhantomData,
28                    _mode: PhantomData,
29                }
30            }
31        }
32        impl<I2C, CONV, MODE> Ads1x1x<I2cInterface<I2C>, ic::$IC, CONV, MODE> {
33            /// Destroy driver instance, return I²C bus instance.
34            pub fn $destroy(self) -> I2C {
35                self.iface.i2c
36            }
37        }
38    };
39}
