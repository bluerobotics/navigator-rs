1//! Common functions

3use crate::{devices::OperatingMode, interface, Ads1x1x, BitFlags, Config, Error, Register};
4
5impl<DI, IC, CONV, MODE, E> Ads1x1x<DI, IC, CONV, MODE>
6where
7    DI: interface::WriteData<Error = E> + interface::ReadData<Error = E>,
8{
9    pub(super) fn set_operating_mode(&mut self, mode: OperatingMode) -> Result<(), Error<E>> {
10        let config;
11        match mode {
12            OperatingMode::OneShot => config = self.config.with_high(BitFlags::OP_MODE),
13            OperatingMode::Continuous => config = self.config.with_low(BitFlags::OP_MODE),
14        }
15        self.iface.write_register(Register::CONFIG, config.bits)?;
16        self.config = config;
17        Ok(())
18    }
19
20    /// Read whether a measurement is currently in progress.
21    pub fn is_measurement_in_progress(&mut self) -> Result<bool, Error<E>> {
22        let config = Config {
23            bits: self.iface.read_register(Register::CONFIG)?,
24        };
25        Ok(!config.is_high(BitFlags::OS))
26    }
27
28    /// Reset the internal state of this driver to the default values.
29    ///
30    /// *Note:* This does not alter the state or configuration of the device.
31    ///
32    /// This resets the cached configuration register value in this driver to
33    /// the power-up (reset) configuration of the device.
34    ///
35    /// This needs to be called after performing a reset on the device, for
36    /// example through an I2C general-call Reset command, which was not done
37    /// through this driver to ensure that the configurations in the device
38    /// and in the driver match.
39    pub fn reset_internal_driver_state(&mut self) {
40        self.config = Config::default();
41    }
42}
