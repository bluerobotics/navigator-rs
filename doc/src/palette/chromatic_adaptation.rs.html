1//! Convert colors from one reference white point to another
2//!
3//! Chromatic adaptation is the human visual system’s ability to adjust to
4//! changes in illumination in order to preserve the appearance of object
5//! colors. It is responsible for the stable appearance of object colours
6//! despite the wide variation of light which might be reflected from an object
7//! and observed by our eyes.
8//!
9//! This library provides three methods for chromatic adaptation Bradford (which
10//! is the default), VonKries and XyzScaling
11//!
12//! ```
13//! use palette::Xyz;
14//! use palette::white_point::{A, C};
15//! use palette::chromatic_adaptation::AdaptInto;
16//!
17//!
18//! let a = Xyz::<A, f32>::with_wp(0.315756, 0.162732, 0.015905);
19//!
20//! //Will convert Xyz<A, f32> to Xyz<C, f32> using Bradford chromatic adaptation
21//! let c: Xyz<C, f32> = a.adapt_into();
22//!
23//! //Should print {x: 0.257963, y: 0.139776,z: 0.058825}
24//! println!("{:?}", c)
25//! ```

33/// Chromatic adaptation methods implemented in the library
34pub enum Method {
35    /// Bradford chromatic adaptation method
36    Bradford,
37    /// VonKries chromatic adaptation method
38    VonKries,
39    /// XyzScaling chromatic adaptation method
40    XyzScaling,
41}

43/// Holds the matrix coefficients for the chromatic adaptation methods
44pub struct ConeResponseMatrices<T: Float> {
45    ///3x3 matrix for the cone response domains
46    pub ma: Mat3<T>,
47    ///3x3 matrix for the inverse of the cone response domains
48    pub inv_ma: Mat3<T>,
49}
50
51/// Generates a conversion matrix to convert the Xyz tristimulus values from
52/// one illuminant to another (Swp -> Dwp)
53pub trait TransformMatrix<Swp, Dwp, T>
54where
55    T: FloatComponent,
56    Swp: WhitePoint,
57    Dwp: WhitePoint,
58{
59    /// Get the cone response functions for the chromatic adaptation method
60    fn get_cone_response(&self) -> ConeResponseMatrices<T>;
61
62    /// Generates a 3x3 transformation matrix to convert color from one
63    /// reference white point to another with the given cone_response
64    fn generate_transform_matrix(&self) -> Mat3<T> {
65        let s_wp: Xyz<Swp, T> = Swp::get_xyz();
66        let t_wp: Xyz<Dwp, T> = Dwp::get_xyz();
67        let adapt = self.get_cone_response();
68
69        let resp_src: Xyz<Swp, _> = multiply_xyz(&adapt.ma, &s_wp);
70        let resp_dst: Xyz<Dwp, _> = multiply_xyz(&adapt.ma, &t_wp);
71        let z = T::zero();
72        let resp = [
73            resp_dst.x / resp_src.x,
74            z,
75            z,
76            z,
77            resp_dst.y / resp_src.y,
78            z,
79            z,
80            z,
81            resp_dst.z / resp_src.z,
82        ];
83
84        let tmp = multiply_3x3(&resp, &adapt.ma);
85        multiply_3x3(&adapt.inv_ma, &tmp)
86    }
87}

144/// Trait to convert color from one reference white point to another
145///
146/// Converts a color from the source white point (Swp) to the destination white
147/// point (Dwp). Uses the bradford method for conversion by default.
148pub trait AdaptFrom<S, Swp, Dwp, T>: Sized
149where
150    T: FloatComponent,
151    Swp: WhitePoint,
152    Dwp: WhitePoint,
153{
154    /// Convert the source color to the destination color using the bradford
155    /// method by default
156    fn adapt_from(color: S) -> Self {
157        Self::adapt_from_using(color, Method::Bradford)
158    }
159    /// Convert the source color to the destination color using the specified
160    /// method
161    fn adapt_from_using<M: TransformMatrix<Swp, Dwp, T>>(color: S, method: M) -> Self;
162}
163
164impl<S, D, Swp, Dwp, T> AdaptFrom<S, Swp, Dwp, T> for D
165where
166    T: FloatComponent,
167    Swp: WhitePoint,
168    Dwp: WhitePoint,
169    S: IntoColorUnclamped<Xyz<Swp, T>>,
170    D: FromColorUnclamped<Xyz<Dwp, T>>,
171{
172    fn adapt_from_using<M: TransformMatrix<Swp, Dwp, T>>(color: S, method: M) -> D {
173        let src_xyz: Xyz<Swp, T> = color.into_color_unclamped();
174        let transform_matrix = method.generate_transform_matrix();
175        let dst_xyz: Xyz<Dwp, T> = multiply_xyz(&transform_matrix, &src_xyz);
176        D::from_color_unclamped(dst_xyz)
177    }
178}

180/// Trait to convert color with one reference white point into another
181///
182/// Converts a color with the source white point (Swp) into the destination
183/// white point (Dwp). Uses the bradford method for conversion by default.
184pub trait AdaptInto<D, Swp, Dwp, T>: Sized
185where
186    T: FloatComponent,
187    Swp: WhitePoint,
188    Dwp: WhitePoint,
189{
190    /// Convert the source color to the destination color using the bradford
191    /// method by default
192    fn adapt_into(self) -> D {
193        self.adapt_into_using(Method::Bradford)
194    }
195    /// Convert the source color to the destination color using the specified
196    /// method
197    fn adapt_into_using<M: TransformMatrix<Swp, Dwp, T>>(self, method: M) -> D;
198}
199
200impl<S, D, Swp, Dwp, T> AdaptInto<D, Swp, Dwp, T> for S
201where
202    T: FloatComponent,
203    Swp: WhitePoint,
204    Dwp: WhitePoint,
205    D: AdaptFrom<S, Swp, Dwp, T>,
206{
207    fn adapt_into_using<M: TransformMatrix<Swp, Dwp, T>>(self, method: M) -> D {
208        D::adapt_from_using(self, method)
209    }
210}
