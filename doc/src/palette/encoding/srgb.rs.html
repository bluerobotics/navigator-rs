11/// The sRGB color space.

13pub struct Srgb;
14
15impl Primaries for Srgb {
16    fn red<Wp: WhitePoint, T: FloatComponent>() -> Yxy<Wp, T> {
17        Yxy::with_wp(from_f64(0.6400), from_f64(0.3300), from_f64(0.212656))
18    }
19    fn green<Wp: WhitePoint, T: FloatComponent>() -> Yxy<Wp, T> {
20        Yxy::with_wp(from_f64(0.3000), from_f64(0.6000), from_f64(0.715158))
21    }
22    fn blue<Wp: WhitePoint, T: FloatComponent>() -> Yxy<Wp, T> {
23        Yxy::with_wp(from_f64(0.1500), from_f64(0.0600), from_f64(0.072186))
24    }
25}

42impl TransferFn for Srgb {
43    fn into_linear<T: Float + FromF64>(x: T) -> T {
44        // Recip call shows performance benefits in benchmarks for this function
45        if x <= from_f64(0.04045) {
46            x * from_f64::<T>(12.92).recip()
47        } else {
48            ((x + from_f64(0.055)) * from_f64::<T>(1.055).recip()).powf(from_f64(2.4))
49        }
50    }
51
52    fn from_linear<T: Float + FromF64>(x: T) -> T {
53        if x <= from_f64(0.0031308) {
54            x * from_f64(12.92)
55        } else {
56            x.powf(T::one() / from_f64(2.4)) * from_f64(1.055) - from_f64(0.055)
57        }
58    }
59}
