9/// Represents colors that can be serialized and deserialized from raw color
10/// components.
11///
12/// This uses bit by bit conversion, so make sure that anything that implements
13/// it can be represented as a contiguous sequence of a single type `T`. This is
14/// most safely done using `#[derive(Pixel)]`.
15///
16/// # Deriving
17///
18/// `Pixel` can be automatically derived. The only requirements are that the
19/// type is a `struct`, that it has a `#[repr(C)]` attribute, and that all of
20/// its fields have the same types. It stays on the conservative side and will
21/// show an error if any of those requirements are not fulfilled. If some fields
22/// have different types, but the same memory layout, or are zero-sized, they
23/// can be marked with attributes to show that their types are safe to use.
24///
25/// ## Field Attributes
26///
27/// * `#[palette_unsafe_same_layout_as = "SomeType"]`: Mark the field as having
28///   the same memory
29/// layout as `SomeType`.
30///
31///   **Unsafety:** corrupt data and undefined behavior may occur if this is not
32/// true!
33///
34/// * `#[palette_unsafe_zero_sized]`: Mark the field as being zero-sized, and
35///   thus not taking up
36/// any memory space. This means that it can be ignored.
37///
38///   **Unsafety:** corrupt data and undefined behavior may occur if this is not
39/// true!
40///
41/// ## Examples
42///
43/// Basic use:
44///
45/// ```rust
46/// use palette::Pixel;
47///
48/// #[derive(PartialEq, Debug, Pixel)]
49/// #[repr(C)]
50/// struct MyCmyk {
51///     cyan: f32,
52///     magenta: f32,
53///     yellow: f32,
54///     key: f32,
55/// }
56///
57/// let buffer = [0.1, 0.2, 0.3, 0.4];
58/// let color = MyCmyk::from_raw(&buffer);
59///
60/// assert_eq!(
61///     color,
62///     &MyCmyk {
63///         cyan: 0.1,
64///         magenta: 0.2,
65///         yellow: 0.3,
66///         key: 0.4,
67///     }
68/// );
69/// ```
70///
71/// Heterogenous field types:
72///
73/// ```rust
74/// use std::marker::PhantomData;
75///
76/// use palette::{Pixel, RgbHue};
77/// use palette::rgb::RgbStandard;
78/// use palette::encoding::Srgb;
79///
80/// #[derive(PartialEq, Debug, Pixel)]
81/// #[repr(C)]
82/// struct MyCoolColor<S: RgbStandard> {
83///     #[palette(unsafe_zero_sized)]
84///     standard: PhantomData<S>,
85///     // RgbHue is a wrapper with `#[repr(C)]`, so it can safely
86///     // be converted straight from `f32`.
87///     #[palette(unsafe_same_layout_as = "f32")]
88///     hue: RgbHue<f32>,
89///     lumen: f32,
90///     chroma: f32,
91/// }
92///
93/// let buffer = [172.0, 100.0, 0.3];
94/// let color = MyCoolColor::<Srgb>::from_raw(&buffer);
95///
96/// assert_eq!(
97///     color,
98///     &MyCoolColor {
99///         hue: 172.0.into(),
100///         lumen: 100.0,
101///         chroma: 0.3,
102///         standard: PhantomData,
103///     }
104/// );
105/// ```
106pub unsafe trait Pixel<T>: Sized {
107    /// The number of color channels.
108    const CHANNELS: usize;
109
110    /// Cast as a reference to raw color components.
111    #[inline]
112    fn as_raw<P: RawPixel<T> + ?Sized>(&self) -> &P {
113        unsafe { P::from_raw_parts(self as *const Self as *const T, Self::CHANNELS) }
114    }
115
116    /// Cast as a mutable reference to raw color components.
117    #[inline]
118    fn as_raw_mut<P: RawPixel<T> + ?Sized>(&mut self) -> &mut P {
119        unsafe { P::from_raw_parts_mut(self as *mut Self as *mut T, Self::CHANNELS) }
120    }
121
122    /// Convert into raw color components.
123    #[inline]
124    fn into_raw<P: RawPixelSized<T>>(self) -> P {
125        assert_eq!(P::CHANNELS, Self::CHANNELS);
126        assert_eq!(::core::mem::size_of::<P>(), ::core::mem::size_of::<Self>());
127        assert_eq!(
128            ::core::mem::align_of::<P>(),
129            ::core::mem::align_of::<Self>()
130        );
131
132        let converted = unsafe { ::core::ptr::read(&self as *const Self as *const P) };
133
134        // Just to be sure...
135        ::core::mem::forget(self);
136
137        converted
138    }
139
140    /// Cast from a reference to raw color components.
141    #[inline]
142    fn from_raw<P: RawPixel<T> + ?Sized>(pixel: &P) -> &Self {
143        assert!(
144            pixel.channels() >= Self::CHANNELS,
145            "not enough color channels"
146        );
147        unsafe { &*(pixel.as_ptr() as *const Self) }
148    }
149
150    /// Cast from a mutable reference to raw color components.
151    #[inline]
152    fn from_raw_mut<P: RawPixel<T> + ?Sized>(pixel: &mut P) -> &mut Self {
153        assert!(pixel.channels() >= Self::CHANNELS);
154        unsafe { &mut *(pixel.as_mut_ptr() as *mut Self) }
155    }
156
157    /// Cast a slice of raw color components to a slice of colors.
158    ///
159    /// ```rust
160    /// use palette::{Pixel, Srgb};
161    ///
162    /// let raw = &[255u8, 128, 64, 10, 20, 30];
163    /// let colors = Srgb::from_raw_slice(raw);
164    ///
165    /// assert_eq!(colors.len(), 2);
166    /// assert_eq!(colors[0].blue, 64);
167    /// assert_eq!(colors[1].red, 10);
168    /// ```
169    #[inline]
170    fn from_raw_slice(slice: &[T]) -> &[Self] {
171        assert_eq!(slice.len() % Self::CHANNELS, 0);
172        let new_length = slice.len() / Self::CHANNELS;
173        unsafe { ::core::slice::from_raw_parts(slice.as_ptr() as *const Self, new_length) }
174    }
175
176    /// Cast a mutable slice of raw color components to a mutable slice of
177    /// colors.
178    ///
179    /// ```rust
180    /// use palette::{Pixel, Srgb};
181    ///
182    /// let raw = &mut [255u8, 128, 64, 10, 20, 30];
183    /// {
184    ///     let colors = Srgb::from_raw_slice_mut(raw);
185    ///     assert_eq!(colors.len(), 2);
186    ///
187    ///     // These changes affects the raw slice, since they are the same data
188    ///     colors[0].blue = 100;
189    ///     colors[1].red = 200;
190    /// }
191    ///
192    /// // Notice the two values in the middle:
193    /// assert_eq!(raw, &[255, 128, 100, 200, 20, 30]);
194    /// ```
195    #[inline]
196    fn from_raw_slice_mut(slice: &mut [T]) -> &mut [Self] {
197        assert_eq!(slice.len() % Self::CHANNELS, 0);
198        let new_length = slice.len() / Self::CHANNELS;
199        unsafe { ::core::slice::from_raw_parts_mut(slice.as_mut_ptr() as *mut Self, new_length) }
200    }
201
202    /// Cast a slice of colors to a slice of raw color components.
203    ///
204    /// ```rust
205    /// use palette::{Pixel, Srgb};
206    ///
207    /// let colors = &[Srgb::new(255u8, 128, 64), Srgb::new(10, 20, 30)];
208    /// let raw = Srgb::into_raw_slice(colors);
209    ///
210    /// assert_eq!(raw.len(), 6);
211    /// assert_eq!(raw, &[255u8, 128, 64, 10, 20, 30]);
212    /// ```
213    #[inline]
214    fn into_raw_slice(slice: &[Self]) -> &[T] {
215        let new_length = slice.len() * Self::CHANNELS;
216        unsafe { ::core::slice::from_raw_parts(slice.as_ptr() as *const T, new_length) }
217    }
218
219    /// Cast a mutable slice of colors to a mutable slice of raw color
220    /// components.
221    ///
222    /// ```rust
223    /// use palette::{Pixel, Srgb};
224    ///
225    /// let colors = &mut [Srgb::new(255u8, 128, 64), Srgb::new(10, 20, 30)];
226    /// {
227    ///     let raw = Srgb::into_raw_slice_mut(colors);
228    ///     assert_eq!(raw.len(), 6);
229    ///
230    ///     // These changes affects the color slice, since they are the same data
231    ///     raw[2] = 100;
232    ///     raw[3] = 200;
233    /// }
234    ///
235    /// assert_eq!(colors[0].blue, 100);
236    /// assert_eq!(colors[1].red, 200);
237    /// ```
238    #[inline]
239    fn into_raw_slice_mut(slice: &mut [Self]) -> &mut [T] {
240        let new_length = slice.len() * Self::CHANNELS;
241        unsafe { ::core::slice::from_raw_parts_mut(slice.as_mut_ptr() as *mut T, new_length) }
242    }
243}
