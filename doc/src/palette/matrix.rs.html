13/// A 9 element array representing a 3x3 matrix.
14pub type Mat3<T> = [T; 9];
15
16/// Multiply the 3x3 matrix with an XYZ color.
17#[inline]
18pub fn multiply_xyz<Swp: WhitePoint, Dwp: WhitePoint, T: FloatComponent>(
19    c: &Mat3<T>,
20    f: &Xyz<Swp, T>,
21) -> Xyz<Dwp, T> {
22    // Input Mat3 is destructured to avoid panic paths
23    let [c0, c1, c2, c3, c4, c5, c6, c7, c8] = *c;
24
25    let x1 = c0 * f.x;
26    let y1 = c3 * f.x;
27    let z1 = c6 * f.x;
28    let x2 = c1 * f.y;
29    let y2 = c4 * f.y;
30    let z2 = c7 * f.y;
31    let x3 = c2 * f.z;
32    let y3 = c5 * f.z;
33    let z3 = c8 * f.z;
34
35    Xyz {
36        x: x1 + x2 + x3,
37        y: y1 + y2 + y3,
38        z: z1 + z2 + z3,
39        white_point: PhantomData,
40    }
41}
42/// Multiply the 3x3 matrix with an XYZ color to return an RGB color.
43#[inline]
44pub fn multiply_xyz_to_rgb<S: RgbSpace, T: FloatComponent>(
45    c: &Mat3<T>,
46    f: &Xyz<S::WhitePoint, T>,
47) -> Rgb<Linear<S>, T> {
48    // Input Mat3 is destructured to avoid panic paths. red, green, and blue
49    // can't be extracted like in `multiply_xyz` to get a performance increase
50    let [c0, c1, c2, c3, c4, c5, c6, c7, c8] = *c;
51
52    Rgb {
53        red: (c0 * f.x) + (c1 * f.y) + (c2 * f.z),
54        green: (c3 * f.x) + (c4 * f.y) + (c5 * f.z),
55        blue: (c6 * f.x) + (c7 * f.y) + (c8 * f.z),
56        standard: PhantomData,
57    }
58}
59/// Multiply the 3x3 matrix with an RGB color to return an XYZ color.
60#[inline]
61pub fn multiply_rgb_to_xyz<S: RgbSpace, T: FloatComponent>(
62    c: &Mat3<T>,
63    f: &Rgb<Linear<S>, T>,
64) -> Xyz<S::WhitePoint, T> {
65    // Input Mat3 is destructured to avoid panic paths. Same problem as
66    // `multiply_xyz_to_rgb` for extracting x, y, z
67    let [c0, c1, c2, c3, c4, c5, c6, c7, c8] = *c;
68
69    Xyz {
70        x: (c0 * f.red) + (c1 * f.green) + (c2 * f.blue),
71        y: (c3 * f.red) + (c4 * f.green) + (c5 * f.blue),
72        z: (c6 * f.red) + (c7 * f.green) + (c8 * f.blue),
73        white_point: PhantomData,
74    }
75}

77/// Multiply two 3x3 matrices.
78#[inline]
79pub fn multiply_3x3<T: Float>(c: &Mat3<T>, f: &Mat3<T>) -> Mat3<T> {
80    // Input Mat3 are destructured to avoid panic paths
81    let [c0, c1, c2, c3, c4, c5, c6, c7, c8] = *c;
82    let [f0, f1, f2, f3, f4, f5, f6, f7, f8] = *f;
83
84    let o0 = c0 * f0 + c1 * f3 + c2 * f6;
85    let o1 = c0 * f1 + c1 * f4 + c2 * f7;
86    let o2 = c0 * f2 + c1 * f5 + c2 * f8;
87
88    let o3 = c3 * f0 + c4 * f3 + c5 * f6;
89    let o4 = c3 * f1 + c4 * f4 + c5 * f7;
90    let o5 = c3 * f2 + c4 * f5 + c5 * f8;
91
92    let o6 = c6 * f0 + c7 * f3 + c8 * f6;
93    let o7 = c6 * f1 + c7 * f4 + c8 * f7;
94    let o8 = c6 * f2 + c7 * f5 + c8 * f8;
95
96    [o0, o1, o2, o3, o4, o5, o6, o7, o8]
97}

99/// Invert a 3x3 matrix and panic if matrix is not invertible.
100#[inline]
101pub fn matrix_inverse<T: Float>(a: &Mat3<T>) -> Mat3<T> {
102    // This function runs fastest with assert and no destructuring. The `det`'s
103    // location should not be changed until benched that it's faster elsewhere
104    assert!(a.len() > 8);
105
106    let d0 = a[4] * a[8] - a[5] * a[7];
107    let d1 = a[3] * a[8] - a[5] * a[6];
108    let d2 = a[3] * a[7] - a[4] * a[6];
109    let mut det = a[0] * d0 - a[1] * d1 + a[2] * d2;
110    let d3 = a[1] * a[8] - a[2] * a[7];
111    let d4 = a[0] * a[8] - a[2] * a[6];
112    let d5 = a[0] * a[7] - a[1] * a[6];
113    let d6 = a[1] * a[5] - a[2] * a[4];
114    let d7 = a[0] * a[5] - a[2] * a[3];
115    let d8 = a[0] * a[4] - a[1] * a[3];
116
117    if !det.is_normal() {
118        panic!("The given matrix is not invertible")
119    }
120    det = det.recip();
121
122    [
123        d0 * det,
124        -d3 * det,
125        d6 * det,
126        -d1 * det,
127        d4 * det,
128        -d7 * det,
129        d2 * det,
130        -d5 * det,
131        d8 * det,
132    ]
133}

135/// Generates the Srgb to Xyz transformation matrix for a given white point.
136#[inline]
137pub fn rgb_to_xyz_matrix<S: RgbSpace, T: FloatComponent>() -> Mat3<T> {
138    let r: Xyz<S::WhitePoint, T> = S::Primaries::red().into_color_unclamped();
139    let g: Xyz<S::WhitePoint, T> = S::Primaries::green().into_color_unclamped();
140    let b: Xyz<S::WhitePoint, T> = S::Primaries::blue().into_color_unclamped();
141
142    // Destructuring has some performance benefits, don't change unless measured
143    let [t0, t1, t2, t3, t4, t5, t6, t7, t8] = mat3_from_primaries(r, g, b);
144
145    let s_matrix: Rgb<Linear<S>, T> = multiply_xyz_to_rgb(
146        &matrix_inverse(&[t0, t1, t2, t3, t4, t5, t6, t7, t8]),
147        &S::WhitePoint::get_xyz(),
148    );
149
150    [
151        t0 * s_matrix.red,
152        t1 * s_matrix.green,
153        t2 * s_matrix.blue,
154        t3 * s_matrix.red,
155        t4 * s_matrix.green,
156        t5 * s_matrix.blue,
157        t6 * s_matrix.red,
158        t7 * s_matrix.green,
159        t8 * s_matrix.blue,
160    ]
161}

163#[rustfmt::skip]
164#[inline]
165fn mat3_from_primaries<T: FloatComponent, Wp: WhitePoint>(r: Xyz<Wp, T>, g: Xyz<Wp, T>, b: Xyz<Wp, T>) -> Mat3<T> {
166    [
167        r.x, g.x, b.x,
168        r.y, g.y, b.y,
169        r.z, g.z, b.z,
170    ]
171}
