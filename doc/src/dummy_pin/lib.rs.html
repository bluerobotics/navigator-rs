1//! This provides dummy implementations of the input/output pin [`embedded-hal`] traits.
2//! This is useful when dealing with setups where a certain pin is handled by hardware in a way
3//! that the software does not need to know about, for example.
4//!
5//! In addition to the no-op, zero-cost `DummyPin`, this crate provides an implementation of `LastStateDummyPin`,
6//! which stores the last state set at runtime and returns it when read.
7//!
8//! [`embedded-hal`]: https://github.com/rust-embedded/embedded-hal
9//!
10//! ## Usage examples
11//!
12//! ### `DummyPin` interaction
13//! A `DummyPin` does nothing and always returns the creation level when read.
14//!
15//! ```
16//! # use embedded_hal::digital::v2::{InputPin, OutputPin, StatefulOutputPin, ToggleableOutputPin};
17//! use dummy_pin::DummyPin;
18//!
19//! // This pin will always read low.
20//! let mut pin = DummyPin::new_low();
21//!
22//! // Using pin as `OutpuPin` does nothing and returns `Ok(())`
23//! pin.set_high().unwrap();
24//!
25//! // Using pin as `InputPin` always returns the creation level.
26//! assert!( pin.is_low().unwrap() );
27//! assert!( !pin.is_high().unwrap() );
28//!
29//! // Using pin as `ToggleableOutputPin` does nothing and returns `Ok(())`
30//! pin.toggle().unwrap();
31//!
32//! // Using pin as `StatefulOutputPin` always returns the creation level.
33//! assert!( pin.is_set_low().unwrap() );
34//! assert!( !pin.is_set_high().unwrap() );
35//! ```
36//!
37//! ### Pass dummy output pin to irrelevant driver output
38//!
39//! ```no_run
40//! use dummy_pin::DummyPin;
41//! use embedded_hal::digital::v2::OutputPin;
42//! use linux_embedded_hal::Pin;
43//!
44//! struct Driver<P> {
45//!     output: P,
46//! }
47//!
48//! impl<P, E> Driver<P>
49//! where
50//!     P: OutputPin<Error = E>,
51//! {
52//!     fn new(pin: P) -> Self {
53//!         Driver { output: pin }
54//!     }
55//!
56//!     fn do_something(&mut self) -> Result<(), E> {
57//!         // ...
58//!         self.output.set_high()
59//!     }
60//! }
61//!
62//! // The same driver can operate with either a real or a dummy pin.
63//! let real_pin = Pin::new(25);
64//! let mut driver_with_real_pin = Driver::new(real_pin);
65//! driver_with_real_pin.do_something().unwrap();
66//!
67//! let dummy_pin = DummyPin::new_low();
68//! let mut driver_with_dummy_pin = Driver::new(dummy_pin);
69//! driver_with_dummy_pin.do_something().unwrap();
70//! ```
71//!
72//! ### `LastStateDummyPin` interaction
73//! A `LastStateDummyPin` stores the last level set and returns it when read.
74//!
75//! ```
76//! # use embedded_hal::digital::v2::{InputPin, OutputPin, StatefulOutputPin, ToggleableOutputPin};
77//! use dummy_pin::LastStateDummyPin;
78//!
79//! // Initially this pin reads low.
80//! let mut pin = LastStateDummyPin::new_low();
81//!
82//! assert!( pin.is_low().unwrap() );
83//!
84//! // Using pin as `OutpuPin` stores the level set
85//! pin.set_high().unwrap();
86//!
87//! // Using pin as `InputPin` returns the last level.
88//! assert!( pin.is_high().unwrap() );
89//!
90//! // Using pin as `ToggleableOutputPin` toggles the internal state
91//! pin.toggle().unwrap();
92//!
93//! assert!( pin.is_low().unwrap() );
94//!
95//! // Using pin as `StatefulOutputPin` returns the last level.
96//! assert!( pin.is_set_low().unwrap() );
97//! assert!( !pin.is_set_high().unwrap() );
98//! ```
99//!
